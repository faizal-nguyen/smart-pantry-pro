# 📋 PRP-032.5 V1 - Mode Expert Optimisé (Drag & Drop + Quick Add)

## 🎯 Vision Produit V1

**Le besoin précis** : Pouvoir planifier sa semaine en glissant ses recettes favorites sur un calendrier, avec vision du budget en temps réel. Alternative : ajouter directement depuis la page recette.

**Benchmark des meilleures solutions** : Après analyse de **Mealime**, **PlateJoy**, **Paprika**, **Plan to Eat** et **BigOven**, voici l'implémentation optimale.

## 🏆 Best Practices du Marché

### Ce qui fonctionne le mieux :

1. **Mealime** : Drag & drop fluide avec preview au survol
2. **Plan to Eat** : Ajout direct depuis la recette avec calendrier popup
3. **Paprika** : Duplication rapide d'une semaine à l'autre
4. **PlateJoy** : Budget en temps réel qui se met à jour
5. **BigOven** : Multi-ajout (même recette plusieurs jours)

// services/meal-planning/planning-service.ts

export class MealPlanningService {
  
  // Duplication intelligente
  async duplicateWeek(fromWeek: Date, toWeek: Date): Promise<void> {
    const meals = await this.getMealsForWeek(fromWeek);
    
    for (const meal of meals) {
      await this.addMeal({
        ...meal,
        week: toWeek,
        id: undefined // Nouveau ID
      });
    }
    
    toast.success('Semaine dupliquée avec succès !');
  }

  // Multi-ajout
  async addRecipeMultipleDays(
    recipeId: string, 
    slots: { day: string, mealType: string }[]
  ): Promise<void> {
    const batch = [];
    
    for (const slot of slots) {
      batch.push(this.addMeal({
        recipeId,
        day: slot.day,
        mealType: slot.mealType
      }));
    }
    
    await Promise.all(batch);
  }

  // Calcul du budget en temps réel
  calculateWeeklyBudget(meals: Meal[]): BudgetBreakdown {
    const total = meals.reduce((sum, meal) => sum + (meal.recipe.cost || 0), 0);
    
    return {
      total,
      perDay: total / 7,
      perMeal: total / meals.length,
      byCategory: this.groupByCategory(meals),
      comparison: this.compareToPreviousWeek(total)
    };
  }

  // Génération de liste de courses optimisée
  async generateShoppingList(weekStart: Date): Promise<ShoppingList> {
    const meals = await this.getMealsForWeek(weekStart);
    const ingredients = this.aggregateIngredients(meals);
    
    // Grouper par rayon
    const byAisle = this.groupByAisle(ingredients);
    
    // Optimiser selon les promos
    const optimized = await this.optimizeWithPromotions(byAisle);
    
    return {
      ingredients: optimized,
      estimatedCost: this.calculateTotalCost(optimized),
      stores: await this.findBestStores(optimized)
    };
  }

  // Suggestions intelligentes basées sur l'historique
  async getSuggestions(slot: MealSlot): Promise<Recipe[]> {
    const context = {
      dayOfWeek: slot.day,
      mealType: slot.mealType,
      season: getCurrentSeason(),
      recentMeals: await this.getRecentMeals(7),
      preferences: await this.getUserPreferences()
    };
    
    // ML pour suggestions personnalisées
    return this.mlRecommendationEngine.suggest(context);
  }

  // Détection de conflits (même repas trop proche)
  detectConflicts(recipe: Recipe, existingMeals: Meal[]): Conflict[] {
    const conflicts = [];
    
    // Même repas dans les 3 jours
    const similar = existingMeals.filter(m => 
      m.recipe.id === recipe.id && 
      daysBetween(m.date, new Date()) < 3
    );
    
    if (similar.length > 0) {
      conflicts.push({
        type: 'repetition',
        message: `${recipe.name} déjà prévu ${similar[0].day}`
      });
    }
    
    return conflicts;
  }
}
