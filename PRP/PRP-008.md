tsxconst OnboardingStep = ({ step, onNext }) => {
  const [answer, setAnswer] = useState(null);
  const controls = useAnimation();
  
  useEffect(() => {
    controls.start({
      opacity: [0, 1],
      y: [20, 0],
      transition: { duration: 0.5 }
    });
  }, [step]);
  
  return (
    <motion.div
      className="min-h-screen flex flex-col justify-center p-6"
      animate={controls}
    >
      {/* Progress Indicator */}
      <div className="fixed top-safe left-0 right-0 px-6">
        <ProgressBar 
          value={step.index} 
          max={totalSteps}
          className="h-1"
        />
      </div>
      
      {/* Content */}
      <div className="flex-1 flex flex-col justify-center">
        {step.type === 'splash' && (
          <SplashStep 
            title={step.content.title}
            subtitle={step.content.subtitle}
            animation={step.content.animation}
          />
        )}
        
        {step.type === 'interactive' && (
          <InteractiveStep
            question={step.question}
            options={step.options}
            value={answer}
            onChange={setAnswer}
            illustration={step.illustration}
          />
        )}
        
        {step.type === 'multi-select' && (
          <MultiSelectStep
            question={step.question}
            options={step.options}
            selected={answer || []}
            onChange={setAnswer}
          />
        )}
      </div>
      
      {/* Actions */}
      <div className="flex gap-3">
        {step.skipable && (
          <button
            onClick={() => onNext(null)}
            className="flex-1 py-4 text-gray-500"
          >
            Passer
          </button>
        )}
        
        <button
          onClick={() => onNext(answer)}
          disabled={!answer && !step.skipable}
          className={cn(
            "flex-1 py-4 rounded-2xl font-semibold transition-all",
            answer || step.skipable
              ? "bg-blue-500 text-white"
              : "bg-gray-100 text-gray-400"
          )}
        >
          {step.index === totalSteps - 1 ? "Commencer" : "Continuer"}
        </button>
      </div>
    </motion.div>
  );
};
