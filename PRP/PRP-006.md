Scanner qui reconnaît instantanément produits, tickets de caisse, et même plats cuisinés.
📷 DESIGN SCANNER
typescriptinterface UltraScanner {
  // 1. MODES DE SCAN
  scanModes: {
    'product': {
      icon: '📦';
      detection: ['barcode', 'ocr', 'image-recognition'];
      feedback: 'real-time-boxes';
      autoCapture: true;
    };
    
    'receipt': {
      icon: '🧾';
      detection: ['ocr', 'table-extraction'];
      guidance: 'frame-overlay';
      multiPage: true;
    };
    
    'meal': {
      icon: '🍽️';
      detection: ['food-recognition', 'portion-estimation'];
      features: ['calorie-estimation', 'recipe-matching'];
    };
    
    'batch': {
      icon: '📦📦';
      detection: ['multi-object'];
      capability: 'scan-multiple-items-at-once';
    };
  };

  // 2. UI AUGMENTÉE
  augmentedUI: {
    liveDetection: {
      boundingBoxes: true;
      labels: true;
      confidence: true;
      hapticOnDetection: true;
    };
    
    guidance: {
      voiceInstructions: true;
      visualGuides: true;
      autoZoom: true;
      lightingTips: true;
    };
  };
}
