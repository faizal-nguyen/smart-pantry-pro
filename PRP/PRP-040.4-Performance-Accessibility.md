### LazyLoadingStrategy.ts
```typescript
interface RouteConfig {
  path: string;
  component: () => Promise<{ default: React.ComponentType }>;
  prefetch?: boolean;
  preload?: boolean;
  critical?: boolean;
}

export class LazyLoadingStrategy {
  private loadedRoutes = new Set<string>();
  private prefetchedRoutes = new Set<string>();
  
  constructor(private routes: RouteConfig[]) {
    this.setupIntersectionObserver();
    this.setupPrefetchOnHover();
  }

  // Lazy loading intelligent avec priorités
  async loadRoute(path: string): Promise<React.ComponentType> {
    if (this.loadedRoutes.has(path)) {
      return this.getFromCache(path);
    }

    const route = this.routes.find(r => r.path === path);
    if (!route) throw new Error(`Route ${path} not found`);

    const startTime = performance.now();
    
    try {
      const module = await route.component();
      const loadTime = performance.now() - startTime;
      
      this.loadedRoutes.add(path);
      this.trackPerformance('route_load', { path, loadTime });
      
      return module.default;
    } catch (error) {
      this.trackError('route_load_failed', { path, error });
      throw error;
    }
  }

  // Prefetch intelligent basé sur probabilités
  async prefetchRoute(path: string, probability = 0.7): Promise<void> {
    if (this.prefetchedRoutes.has(path) || probability < 0.5) return;

    const route = this.routes.find(r => r.path === path);
    if (!route) return;

    requestIdleCallback(async () => {
      try {
        await route.component();
        this.prefetchedRoutes.add(path);
        this.trackPerformance('route_prefetch', { path, probability });
      } catch (error) {
        this.trackError('prefetch_failed', { path, error });
      }
    });
  }

  private setupPrefetchOnHover(): void {
    document.addEventListener('mouseover', (event) => {
      const link = (event.target as Element)?.closest('[data-prefetch]');
      if (link) {
        const path = link.getAttribute('data-prefetch');
        if (path) this.prefetchRoute(path, 0.8);
      }
    });
  }
}
```

### ScreenReaderAnnouncer.ts
```typescript
export class ScreenReaderAnnouncer {
  private liveRegion: HTMLElement;
  private politeRegion: HTMLElement;

  constructor() {
    this.createLiveRegions();
  }

  // Annonce immédiate (assertive)
  announceImmediate(message: string): void {
    this.liveRegion.textContent = message;
    
    // Clear after announcement
    setTimeout(() => {
      this.liveRegion.textContent = '';
    }, 1000);
  }

  // Annonce polie (attendre pause)
  announcePolite(message: string): void {
    this.politeRegion.textContent = message;
    
    setTimeout(() => {
      this.politeRegion.textContent = '';
    }, 3000);
  }

  // Annonce navigation
  announceNavigation(from: string, to: string): void {
    const message = `Navigation de ${from} vers ${to}`;
    this.announcePolite(message);
  }

  // Annonce état page
  announcePageLoad(title: string, itemCount?: number): void {
    let message = `Page ${title} chargée`;
    if (itemCount) message += `, ${itemCount} éléments disponibles`;
    
    this.announcePolite(message);
  }

  private createLiveRegions(): void {
    // Région assertive
    this.liveRegion = document.createElement('div');
    this.liveRegion.setAttribute('aria-live', 'assertive');
    this.liveRegion.setAttribute('aria-atomic', 'true');
    this.liveRegion.className = 'sr-only';
    
    // Région polie
    this.politeRegion = document.createElement('div');
    this.politeRegion.setAttribute('aria-live', 'polite');
    this.politeRegion.setAttribute('aria-atomic', 'true');
    this.politeRegion.className = 'sr-only';
    
    document.body.appendChild(this.liveRegion);
    document.body.appendChild(this.politeRegion);
  }
}
```

### FocusManager.ts
```typescript
export class FocusManager {
  private focusHistory: HTMLElement[] = [];
  private currentTrap: HTMLElement | null = null;

  // Focus management avec historique
  manageFocus(element: HTMLElement, options: {
    preventScroll?: boolean;
    restoreOnUnmount?: boolean;
  } = {}): void {
    if (document.activeElement !== element) {
      this.focusHistory.push(document.activeElement as HTMLElement);
    }

    element.focus({ preventScroll: options.preventScroll });
    
    if (options.restoreOnUnmount) {
      this.setupRestoreOnUnmount(element);
    }
  }

  // Focus trap pour modales
  trapFocus(container: HTMLElement): void {
    this.currentTrap = container;
    const focusableElements = this.getFocusableElements(container);
    
    if (focusableElements.length === 0) return;

    // Focus premier élément
    focusableElements[0].focus();

    // Gestion Tab cycling
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key !== 'Tab') return;
      
      const currentIndex = focusableElements.indexOf(document.activeElement as HTMLElement);
      
      if (e.shiftKey) {
        // Shift+Tab - previous
        const nextIndex = currentIndex <= 0 ? focusableElements.length - 1 : currentIndex - 1;
        focusableElements[nextIndex].focus();
      } else {
        // Tab - next
        const nextIndex = currentIndex >= focusableElements.length - 1 ? 0 : currentIndex + 1;
        focusableElements[nextIndex].focus();
      }
      
      e.preventDefault();
    };

    container.addEventListener('keydown', handleKeyDown);
    
    // Cleanup function
    return () => {
      container.removeEventListener('keydown', handleKeyDown);
      this.currentTrap = null;
    };
  }

  // Retour focus précédent
  restorePreviousFocus(): void {
    const previous = this.focusHistory.pop();
    if (previous && document.contains(previous)) {
      previous.focus();
    }
  }

  // Skip links pour navigation rapide
  createSkipLinks(): HTMLElement {
    const skipContainer = document.createElement('div');
    skipContainer.className = 'skip-links';
    
    const skipLinks = [
      { href: '#main-content', text: 'Aller au contenu principal' },
      { href: '#navigation', text: 'Aller à la navigation' },
      { href: '#search', text: 'Aller à la recherche' }
    ];

    skipLinks.forEach(link => {
      const skipLink = document.createElement('a');
      skipLink.href = link.href;
      skipLink.textContent = link.text;
      skipLink.className = 'skip-link';
      skipContainer.appendChild(skipLink);
    });

    return skipContainer;
  }

  private getFocusableElements(container: HTMLElement): HTMLElement[] {
    const selectors = [
      'button:not([disabled])',
      'input:not([disabled])',
      'select:not([disabled])',
      'textarea:not([disabled])',
      'a[href]',
      '[tabindex]:not([tabindex="-1"])'
    ];

    return Array.from(container.querySelectorAll(selectors.join(',')))
      .filter(el => !el.hasAttribute('inert')) as HTMLElement[];
  }
}
```

### Performance Budgets & Monitoring
```typescript
export class PerformanceMonitor {
  private metrics: PerformanceMetrics = {
    navigationTiming: [],
    routeLoadTime: new Map(),
    memoryUsage: [],
    renderMetrics: []
  };

  // Monitoring continu Core Web Vitals
  startMonitoring(): void {
    // Largest Contentful Paint
    this.observeLCP();
    
    // First Input Delay
    this.observeFID();
    
    // Cumulative Layout Shift
    this.observeCLS();
    
    // Navigation timing
    this.observeNavigationTiming();
    
    // Memory usage
    this.monitorMemoryUsage();
  }

  private observeLCP(): void {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      const lcp = entries[entries.length - 1];
      
      this.trackMetric('lcp', lcp.startTime);
      
      if (lcp.startTime > 2500) {
        this.reportPerformanceIssue('LCP_THRESHOLD_EXCEEDED', {
          value: lcp.startTime,
          threshold: 2500
        });
      }
    }).observe({ entryTypes: ['largest-contentful-paint'] });
  }

  private observeFID(): void {
    new PerformanceObserver((entryList) => {
      entryList.getEntries().forEach(entry => {
        this.trackMetric('fid', entry.processingStart - entry.startTime);
        
        if (entry.processingStart - entry.startTime > 100) {
          this.reportPerformanceIssue('FID_THRESHOLD_EXCEEDED', {
            value: entry.processingStart - entry.startTime,
            threshold: 100
          });
        }
      });
    }).observe({ entryTypes: ['first-input'] });
  }

  // Performance budget enforcement
  enforcePerformanceBudgets(): void {
    const budgets = {
      bundleSize: 250000, // 250KB
      routeLoadTime: 200, // 200ms
      memoryUsage: 50000000 // 50MB
    };

    // Check bundle size
    if (this.getCurrentBundleSize() > budgets.bundleSize) {
      console.warn('Bundle size budget exceeded');
      this.reportBudgetViolation('bundle_size', this.getCurrentBundleSize(), budgets.bundleSize);
    }

    // Check average route load time
    const avgLoadTime = this.getAverageRouteLoadTime();
    if (avgLoadTime > budgets.routeLoadTime) {
      console.warn('Route load time budget exceeded');
      this.reportBudgetViolation('route_load_time', avgLoadTime, budgets.routeLoadTime);
    }
  }
}
```

```typescript
// tests/accessibility/navigation.test.ts
import { render, screen } from '@testing-library/react';
import { axe, toHaveNoViolations } from 'jest-axe';
import { AppNavigation } from '@/components/navigation/AppNavigation';

expect.extend(toHaveNoViolations);

describe('Navigation Accessibility', () => {
  test('should have no accessibility violations', async () => {
    const { container } = render(<AppNavigation />);
    const results = await axe(container);
    expect(results).toHaveNoViolations();
  });

  test('should support keyboard navigation', async () => {
    render(<AppNavigation />);
    
    const firstNavItem = screen.getByRole('button', { name: /garde-manger/i });
    firstNavItem.focus();
    
    expect(firstNavItem).toHaveFocus();
    
    fireEvent.keyDown(firstNavItem, { key: 'Tab' });
    expect(screen.getByRole('button', { name: /cuisine/i })).toHaveFocus();
  });

  test('should announce navigation changes to screen readers', async () => {
    render(<AppNavigation />);
    
    const mockAnnounce = jest.spyOn(ScreenReaderAnnouncer.prototype, 'announceNavigation');
    
    fireEvent.click(screen.getByRole('button', { name: /recettes/i }));
    
    expect(mockAnnounce).toHaveBeenCalledWith('garde-manger', 'recettes');
  });

  test('should respect reduced motion preferences', () => {
    // Mock prefers-reduced-motion
    Object.defineProperty(window, 'matchMedia', {
      writable: true,
      value: jest.fn().mockImplementation(query => ({
        matches: query === '(prefers-reduced-motion: reduce)',
        media: query,
        onchange: null,
        addListener: jest.fn(),
        removeListener: jest.fn(),
      })),
    });

    render(<AppNavigation />);
    
    const navElement = screen.getByRole('navigation');
    expect(navElement).toHaveClass('motion-reduced');
  });
});
```
