### CommandPalette.tsx
```typescript
interface CommandPaletteProps {
  open: boolean;
  onOpenChange: (open: boolean) => void;
}

export function CommandPalette({ open, onOpenChange }: CommandPaletteProps) {
  const navigate = useNavigate();
  const location = useLocation();

  const commands = useMemo(() => [
    // Navigation rapide
    {
      group: "Navigation",
      items: [
        { id: "go-pantry", label: "Aller au garde-manger", icon: Package, action: () => navigate("/pantry") },
        { id: "go-recipes", label: "Voir les recettes", icon: ChefHat, action: () => navigate("/kitchen/recipes") },
        // ...
      ]
    },
    // Actions rapides
    {
      group: "Actions",
      items: [
        { id: "add-product", label: "Ajouter un produit", icon: Plus, action: openAddProductModal },
        { id: "scan-product", label: "Scanner un produit", icon: Camera, action: () => navigate("/pantry/scanner") },
        // ...
      ]
    },
    // Actions contextuelles
    {
      group: "Contexte",
      items: getContextualActions(location.pathname)
    }
  ], [location.pathname, navigate]);

  return (
    <CommandDialog open={open} onOpenChange={onOpenChange}>
      <CommandInput placeholder="Que souhaitez-vous faire ?" />
      <CommandList>
        <CommandEmpty>Aucun résultat trouvé.</CommandEmpty>
        {commands.map((group) => (
          <CommandGroup key={group.group} heading={group.group}>
            {group.items.map((item) => (
              <CommandItem
                key={item.id}
                onSelect={() => {
                  item.action();
                  onOpenChange(false);
                  trackCommandUsage(item.id);
                }}
              >
                <item.icon className="mr-2 h-4 w-4" />
                <span>{item.label}</span>
              </CommandItem>
            ))}
          </CommandGroup>
        ))}
      </CommandList>
    </CommandDialog>
  );
}
```

### QuickActionsFAB.tsx
```typescript
interface FABAction {
  id: string;
  icon: React.ElementType;
  label: string;
  action: () => void;
  primary?: boolean;
}

export function QuickActionsFAB() {
  const location = useLocation();
  const { isMobile } = useBreakpoints();
  const [expanded, setExpanded] = useState(false);

  const getContextualActions = (pathname: string): FABAction[] => {
    switch (true) {
      case pathname.includes('/pantry'):
        return [
          { id: 'scan', icon: Camera, label: 'Scanner', action: () => navigate('/pantry/scanner'), primary: true },
          { id: 'add', icon: Plus, label: 'Ajouter', action: openAddProductModal },
          { id: 'search', icon: Search, label: 'Chercher', action: openSearchModal }
        ];
      case pathname.includes('/kitchen'):
        return [
          { id: 'timer', icon: Clock, label: 'Timer', action: startCookingTimer, primary: true },
          { id: 'recipe', icon: Plus, label: 'Nouvelle recette', action: () => navigate('/kitchen/recipes/new') },
          { id: 'suggest', icon: Sparkles, label: 'Suggérer', action: openRecipeSuggestions }
        ];
      default:
        return [
          { id: 'scan', icon: Camera, label: 'Scanner', action: () => navigate('/pantry/scanner'), primary: true },
          { id: 'search', icon: Search, label: 'Rechercher', action: openGlobalSearch }
        ];
    }
  };

  const actions = getContextualActions(location.pathname);
  const primaryAction = actions.find(a => a.primary) || actions[0];

  if (!isMobile) return null;

  return (
    <div className="fixed bottom-20 right-4 z-50">
      {/* Actions secondaires */}
      {expanded && (
        <div className="mb-3 flex flex-col gap-2">
          {actions.filter(a => !a.primary).map((action, index) => (
            <motion.div
              key={action.id}
              initial={{ scale: 0, y: 20 }}
              animate={{ scale: 1, y: 0 }}
              transition={{ delay: index * 0.1 }}
            >
              <Button
                size="icon"
                variant="secondary"
                className="h-12 w-12 rounded-full shadow-lg"
                onClick={() => {
                  action.action();
                  setExpanded(false);
                  trackFABUsage(action.id);
                }}
              >
                <action.icon className="h-5 w-5" />
              </Button>
            </motion.div>
          ))}
        </div>
      )}

      {/* FAB principal */}
      <Button
        size="icon"
        className="h-14 w-14 rounded-full shadow-lg"
        onTouchStart={() => setExpanded(true)}
        onClick={primaryAction.action}
      >
        <primaryAction.icon className="h-6 w-6" />
      </Button>
    </div>
  );
}
```
