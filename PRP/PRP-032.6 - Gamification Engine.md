// services/gamification/daily-feast-engine.ts

import { EventEmitter } from 'events';
import { supabase } from '@/lib/supabase';
import { analytics } from '@/services/analytics';

interface DailyFeastConfig {
  streakMultiplier: number;
  heartsRefillRate: number; // minutes
  questXPBase: number;
  gardenGrowthRate: number;
}

export class DailyFeastEngine extends EventEmitter {
  private config: DailyFeastConfig;
  private userId: string;
  private cache: Map<string, any> = new Map();

  constructor(userId: string) {
    super();
    this.userId = userId;
    this.config = {
      streakMultiplier: 1.1,
      heartsRefillRate: 240, // 4 hours
      questXPBase: 100,
      gardenGrowthRate: 1 // 1 vegetable per day
    };
    this.initialize();
  }

  private async initialize() {
    await this.loadUserState();
    this.startHeartRefillTimer();
    this.checkDailyReset();
    this.generateDailyQuests();
  }

  // ============= STREAK SYSTEM =============

  async maintainStreak(): Promise<StreakResult> {
    const { data: streak } = await supabase
      .from('user_streaks')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const today = new Date().toDateString();
    const lastActivity = streak?.last_activity_date 
      ? new Date(streak.last_activity_date).toDateString() 
      : null;

    if (lastActivity === today) {
      return { maintained: true, current: streak.current_streak };
    }

    const yesterday = new Date(Date.now() - 86400000).toDateString();
    let newStreak = streak?.current_streak || 0;
    let freezeUsed = false;

    if (lastActivity === yesterday) {
      // Continue streak
      newStreak += 1;
    } else if (streak?.streak_freeze_available > 0 && lastActivity) {
      // Use freeze
      freezeUsed = true;
      await this.useStreakFreeze();
    } else {
      // Break streak
      newStreak = 1;
      await this.handleStreakBreak(streak?.current_streak || 0);
    }

    // Update streak
    await supabase
      .from('user_streaks')
      .upsert({
        user_id: this.userId,
        current_streak: newStreak,
        longest_streak: Math.max(newStreak, streak?.longest_streak || 0),
        last_activity_date: new Date().toISOString().split('T')[0],
        total_days_active: (streak?.total_days_active || 0) + 1
      });

    // Check mini-streaks
    await this.updateMiniStreaks();

    // Fire events
    this.emit('streak_maintained', {
      current: newStreak,
      freezeUsed,
      milestone: this.checkStreakMilestone(newStreak)
    });

    // Analytics
    analytics.track('streak_maintained', {
      user_id: this.userId,
      streak_length: newStreak,
      freeze_used: freezeUsed
    });

    return {
      maintained: true,
      current: newStreak,
      freezeUsed,
      xpEarned: this.calculateStreakXP(newStreak)
    };
  }

  private async updateMiniStreaks() {
    const today = new Date().getDay();
    const updates: any = {};

    // Veggie Monday
    if (today === 1) { // Monday
      const hasVeggieMeal = await this.checkVeggieMealToday();
      if (hasVeggieMeal) {
        updates.veggie_monday_streak = supabase.raw('veggie_monday_streak + 1');
      } else {
        updates.veggie_monday_streak = 0;
      }
    }

    // Budget Hero
    const underBudget = await this.checkBudgetToday();
    if (underBudget) {
      updates.budget_hero_streak = supabase.raw('budget_hero_streak + 1');
    } else {
      updates.budget_hero_streak = 0;
    }

    // Quick Chef
    const quickMeals = await this.checkQuickMealsToday();
    if (quickMeals) {
      updates.quick_chef_streak = supabase.raw('quick_chef_streak + 1');
    } else {
      updates.quick_chef_streak = 0;
    }

    if (Object.keys(updates).length > 0) {
      await supabase
        .from('user_streaks')
        .update(updates)
        .eq('user_id', this.userId);
    }
  }

  private calculateStreakXP(streak: number): number {
    // Exponential growth with milestones
    const base = 10;
    const milestoneBonus = [7, 14, 30, 60, 100, 365];
    
    let xp = base * Math.ceil(Math.log2(streak + 1));
    
    if (milestoneBonus.includes(streak)) {
      xp += streak * 10; // Big bonus at milestones
    }
    
    return Math.round(xp);
  }

  // ============= HEARTS SYSTEM =============

  async loseHeart(reason: string): Promise<HeartResult> {
    const { data: hearts } = await supabase
      .from('user_hearts')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    if (hearts?.unlimited_hearts) {
      return { heartsRemaining: -1, unlimited: true };
    }

    const currentHearts = hearts?.current_hearts || 5;
    
    if (currentHearts <= 0) {
      this.emit('no_hearts_remaining');
      return { heartsRemaining: 0, blocked: true };
    }

    const newHearts = currentHearts - 1;
    
    await supabase
      .from('user_hearts')
      .upsert({
        user_id: this.userId,
        current_hearts: newHearts,
        last_heart_lost_at: new Date(),
        hearts_lost_reasons: supabase.raw(`
          hearts_lost_reasons || jsonb_build_array(
            jsonb_build_object(
              'reason', '${reason}',
              'timestamp', to_jsonb(now())
            )
          )
        `)
      });

    if (newHearts === 0) {
      this.emit('hearts_depleted');
      this.scheduleHeartRefill();
    }

    analytics.track('heart_lost', {
      user_id: this.userId,
      reason,
      hearts_remaining: newHearts
    });

    return { heartsRemaining: newHearts };
  }

  private startHeartRefillTimer() {
    setInterval(async () => {
      const { data: hearts } = await supabase
        .from('user_hearts')
        .select('*')
        .eq('user_id', this.userId)
        .single();

      if (!hearts || hearts.unlimited_hearts) return;

      const timeSinceLastLoss = Date.now() - new Date(hearts.last_heart_lost_at).getTime();
      const refillTime = this.config.heartsRefillRate * 60 * 1000;

      if (hearts.current_hearts < hearts.max_hearts && timeSinceLastLoss >= refillTime) {
        await supabase
          .from('user_hearts')
          .update({
            current_hearts: Math.min(hearts.current_hearts + 1, hearts.max_hearts),
            last_refill_at: new Date()
          })
          .eq('user_id', this.userId);

        this.emit('heart_refilled', {
          current: hearts.current_hearts + 1,
          max: hearts.max_hearts
        });
      }
    }, 60000); // Check every minute
  }

  // ============= DAILY QUESTS =============

  async generateDailyQuests(): Promise<DailyQuests> {
    const userLevel = await this.getUserLevel();
    const questDate = new Date().toISOString().split('T')[0];

    // Check if already generated
    const existing = await supabase
      .from('daily_quests')
      .select('*')
      .eq('user_id', this.userId)
      .eq('quest_date', questDate)
      .single();

    if (existing.data) {
      return existing.data;
    }

    // Generate new quests based on level and patterns
    const mainQuest = this.generateMainQuest(userLevel);
    const bonusQuests = this.generateBonusQuests(userLevel);
    const superQuest = this.generateSuperQuest();

    const quests = {
      user_id: this.userId,
      quest_date: questDate,
      main_quest: mainQuest,
      bonus_quests: bonusQuests,
      super_quest: superQuest
    };

    await supabase
      .from('daily_quests')
      .insert(quests);

    this.emit('quests_generated', quests);

    return quests;
  }

  private generateMainQuest(level: string): Quest {
    const questTemplates = {
      beginner: [
        { type: 'plan_meals', target: 3, xp: 100 },
        { type: 'add_vegetarian', target: 1, xp: 100 },
        { type: 'stay_under_budget', target: 2, xp: 100 }
      ],
      organized: [
        { type: 'plan_meals', target: 5, xp: 150 },
        { type: 'batch_cooking', target: 1, xp: 150 },
        { type: 'meal_prep', target: 3, xp: 150 }
      ],
      advanced: [
        { type: 'plan_week', target: 7, xp: 200 },
        { type: 'zero_waste', target: 1, xp: 200 },
        { type: 'new_cuisine', target: 2, xp: 200 }
      ]
    };

    const templates = questTemplates[level] || questTemplates.beginner;
    return {
      ...templates[Math.floor(Math.random() * templates.length)],
      progress: 0,
      completed: false
    };
  }

  private generateBonusQuests(level: string): Quest[] {
    const allBonusQuests = [
      { type: 'add_seasonal', target: 1, xp: 50 },
      { type: 'under_5_euro', target: 1, xp: 50 },
      { type: 'quick_meal', target: 1, xp: 50 },
      { type: 'share_recipe', target: 1, xp: 50 },
      { type: 'take_photo', target: 1, xp: 30 },
      { type: 'rate_meal', target: 2, xp: 40 },
      { type: 'local_ingredient', target: 1, xp: 60 }
    ];

    // Select 3-4 random bonus quests
    const count = Math.random() > 0.5 ? 4 : 3;
    const selected = [];
    const available = [...allBonusQuests];

    for (let i = 0; i < count; i++) {
      const index = Math.floor(Math.random() * available.length);
      selected.push({
        ...available[index],
        progress: 0,
        completed: false
      });
      available.splice(index, 1);
    }

    return selected;
  }

  async updateQuestProgress(questType: string, increment: number = 1): Promise<QuestUpdate> {
    const questDate = new Date().toISOString().split('T')[0];
    
    const { data: quests } = await supabase
      .from('daily_quests')
      .select('*')
      .eq('user_id', this.userId)
      .eq('quest_date', questDate)
      .single();

    if (!quests) return { updated: false };

    let xpEarned = 0;
    let questCompleted = false;
    const updates: any = {};

    // Check main quest
    if (quests.main_quest.type === questType && !quests.main_quest.completed) {
      quests.main_quest.progress += increment;
      if (quests.main_quest.progress >= quests.main_quest.target) {
        quests.main_quest.completed = true;
        questCompleted = true;
        xpEarned += quests.main_quest.xp;
      }
      updates.main_quest = quests.main_quest;
    }

    // Check bonus quests
    const updatedBonus = quests.bonus_quests.map(quest => {
      if (quest.type === questType && !quest.completed) {
        quest.progress += increment;
        if (quest.progress >= quest.target) {
          quest.completed = true;
          questCompleted = true;
          xpEarned += quest.xp;
        }
      }
      return quest;
    });
    updates.bonus_quests = updatedBonus;

    // Check if all quests completed for super quest
    const allCompleted = quests.main_quest.completed && 
                        updatedBonus.every(q => q.completed);

    if (allCompleted && !quests.all_completed) {
      updates.all_completed = true;
      updates.super_quest = {
        ...quests.super_quest,
        unlocked: true
      };
      this.emit('super_quest_unlocked');
    }

    // Update database
    await supabase
      .from('daily_quests')
      .update({
        ...updates,
        total_xp_earned: (quests.total_xp_earned || 0) + xpEarned
      })
      .eq('user_id', this.userId)
      .eq('quest_date', questDate);

    if (questCompleted) {
      this.emit('quest_completed', {
        type: questType,
        xpEarned
      });
    }

    return {
      updated: true,
      questCompleted,
      xpEarned,
      allQuestsCompleted: allCompleted
    };
  }

  // ============= LEARNING PATH =============

  async unlockSkill(skillId: string): Promise<SkillUnlock> {
    const { data: progress } = await supabase
      .from('skill_tree_progress')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const skillTree = this.getSkillTree();
    const skill = skillTree[skillId];

    if (!skill) {
      throw new Error('Skill not found');
    }

    // Check prerequisites
    if (skill.prerequisites) {
      const hasAllPrereqs = skill.prerequisites.every(prereq => 
        progress?.completed_skills?.includes(prereq)
      );
      
      if (!hasAllPrereqs) {
        return { 
          unlocked: false, 
          reason: 'Missing prerequisites' 
        };
      }
    }

    // Check skill points
    if ((progress?.skill_points || 0) < skill.cost) {
      return {
        unlocked: false,
        reason: 'Insufficient skill points'
      };
    }

    // Unlock skill
    await supabase
      .from('skill_tree_progress')
      .update({
        unlocked_skills: supabase.raw(`
          array_append(unlocked_skills, '${skillId}')
        `),
        skill_points: (progress?.skill_points || 0) - skill.cost,
        skills_in_progress: {
          ...progress?.skills_in_progress,
          [skillId]: {
            progress: 0,
            total: skill.lessons.length,
            current_lesson: 0
          }
        }
      })
      .eq('user_id', this.userId);

    this.emit('skill_unlocked', {
      skillId,
      skillName: skill.name
    });

    return {
      unlocked: true,
      skill,
      nextLesson: skill.lessons[0]
    };
  }

  async completeLesson(skillId: string, lessonId: string): Promise<LessonCompletion> {
    const { data: progress } = await supabase
      .from('skill_tree_progress')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const skillProgress = progress?.skills_in_progress?.[skillId];
    if (!skillProgress) {
      throw new Error('Skill not in progress');
    }

    const skill = this.getSkillTree()[skillId];
    const lessonIndex = skill.lessons.findIndex(l => l.id === lessonId);

    if (lessonIndex !== skillProgress.current_lesson) {
      throw new Error('Not the current lesson');
    }

    // Update progress
    const newProgress = skillProgress.progress + 1;
    const isSkillComplete = newProgress >= skill.lessons.length;

    const updates: any = {
      skills_in_progress: {
        ...progress.skills_in_progress,
        [skillId]: {
          ...skillProgress,
          progress: newProgress,
          current_lesson: isSkillComplete ? -1 : lessonIndex + 1
        }
      },
      total_lessons_completed: (progress?.total_lessons_completed || 0) + 1
    };

    if (isSkillComplete) {
      updates.completed_skills = supabase.raw(`
        array_append(completed_skills, '${skillId}')
      `);
      updates.skill_points = (progress?.skill_points || 0) + skill.completion_reward;
    }

    await supabase
      .from('skill_tree_progress')
      .update(updates)
      .eq('user_id', this.userId);

    // Award XP
    const xpEarned = skill.lessons[lessonIndex].duration * 10;
    await this.awardXP(xpEarned, 'lesson_completed');

    this.emit('lesson_completed', {
      skillId,
      lessonId,
      xpEarned,
      skillComplete: isSkillComplete
    });

    return {
      completed: true,
      xpEarned,
      skillComplete: isSkillComplete,
      nextLesson: isSkillComplete ? null : skill.lessons[lessonIndex + 1]
    };
  }

    const vegetableTypes = ['carrot', 'tomato', 'lettuce', 'cucumber', 'pepper', 'corn', 'broccoli'];
    const newVegetable = {
      type: vegetableTypes[Math.floor(Math.random() * vegetableTypes.length)],
      planted_date: new Date().toISOString(),
      grown: false,
      position: garden?.personal_vegetables || 0
    };

    const updatedLayout = [...(garden?.personal_garden_layout || []), newVegetable];

    await supabase
      .from('user_garden')
      .upsert({
        user_id: this.userId,
        household_id: this.householdId,
        personal_vegetables: (garden?.personal_vegetables || 0) + 1,
        personal_garden_layout: updatedLayout,
        family_contribution: (garden?.family_contribution || 0) + 1
      });

    // Check for harvest (every 7 vegetables)
    if ((garden?.personal_vegetables || 0) % 7 === 6) {
      await this.harvestGarden();
    }

    this.emit('vegetable_planted', {
      type: newVegetable.type,
      total: (garden?.personal_vegetables || 0) + 1
    });

    return {
      planted: newVegetable,
      totalVegetables: (garden?.personal_vegetables || 0) + 1,
      readyToHarvest: (garden?.personal_vegetables || 0) % 7 === 6
    };
  }

  private async harvestGarden(): Promise<HarvestResult> {
    const { data: garden } = await supabase
      .from('user_garden')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const harvestRewards = {
      xp: 200,
      recipes: ['seasonal_salad', 'vegetable_soup'],
      badge: 'green_thumb'
    };

    // Mark vegetables as harvested
    const updatedLayout = garden.personal_garden_layout.map((veg, index) => {
      if (index < garden.personal_garden_layout.length - 7) {
        return veg;
      }
      return { ...veg, grown: true };
    });

    await supabase
      .from('user_garden')
      .update({
        personal_garden_layout: updatedLayout,
        last_harvest_date: new Date().toISOString().split('T')[0],
        total_harvests: (garden?.total_harvests || 0) + 1
      })
      .eq('user_id', this.userId);

    // Award rewards
    await this.awardXP(harvestRewards.xp, 'garden_harvest');
    await this.unlockRecipes(harvestRewards.recipes);

    this.emit('garden_harvested', harvestRewards);

    return harvestRewards;
  }

  // ============= MICRO-ACHIEVEMENTS =============

  async checkMicroAchievements(context: any): Promise<Achievement[]> {
    const unlocked = [];
    const hour = new Date().getHours();
    const day = new Date().getDay();

    // Early Bird
    if (hour < 9 && context.action === 'plan_meal') {
      const achievement = await this.unlockMicroAchievement('early_bird', 25);
      if (achievement) unlocked.push(achievement);
    }

    // Night Owl
    if (hour >= 22 && context.action === 'plan_meal') {
      const achievement = await this.unlockMicroAchievement('night_owl', 30);
      if (achievement) unlocked.push(achievement);
    }

    // Speed Demon
    if (context.time_taken < 30 && context.action === 'complete_planning') {
      const achievement = await this.unlockMicroAchievement('speed_demon', 40);
      if (achievement) unlocked.push(achievement);
    }

    // Combo Master
    if (context.consecutive_actions >= 3) {
      const achievement = await this.unlockMicroAchievement('combo_master', 35);
      if (achievement) unlocked.push(achievement);
    }

    // Weekend Warrior
    if ((day === 0 || day === 6) && context.action === 'plan_week') {
      const achievement = await this.unlockMicroAchievement('weekend_warrior', 50);
      if (achievement) unlocked.push(achievement);
    }

    return unlocked;
  }

  private async unlockMicroAchievement(type: string, xp: number): Promise<Achievement> {
    // Check if already unlocked today
    const today = new Date().toISOString().split('T')[0];
    const { data: existing } = await supabase
      .from('micro_achievements')
      .select('*')
      .eq('user_id', this.userId)
      .eq('achievement_type', type)
      .gte('unlocked_at', today)
      .single();

    if (existing) return null;

    const achievement = {
      user_id: this.userId,
      achievement_type: type,
      achievement_name: this.getAchievementName(type),
      xp_earned: xp,
      rarity: this.getAchievementRarity(type),
      day_of_week: new Date().getDay(),
      time_of_day: new Date().toTimeString().split(' ')[0]
    };

    await supabase
      .from('micro_achievements')
      .insert(achievement);

    await this.awardXP(xp, `achievement_${type}`);

    this.emit('micro_achievement_unlocked', achievement);

    return achievement;
  }

  private generateSmartSuggestion(context: any): CoachMessage {
    const suggestions = [];

    // Variety suggestion
    if (this.needsMoreVariety(context.recentMeals)) {
      suggestions.push({
        type: 'variety',
        message: "Je vois que tu aimes les pâtes ! Essayons un curry thaï aujourd'hui ?",
        data: { suggested_cuisine: 'thai', reason: 'variety' }
      });
    }

    // Budget suggestion
    if (this.isOverBudget(context.recentMeals)) {
      suggestions.push({
        type: 'budget',
        message: "Astuce : Les légumes de saison coûtent 30% moins cher cette semaine",
        data: { seasonal_vegetables: ['courge', 'poireau', 'chou'] }
      });
    }

    // Time suggestion
    if (context.timeOfDay >= 18 && context.timeOfDay <= 20) {
      suggestions.push({
        type: 'time',
        message: "Repas rapide pour ce soir ? Cette salade complète prend 15 minutes",
        data: { quick_recipe: 'caesar_salad_express' }
      });
    }

    // Health suggestion
    if (this.needsMoreVegetables(context.recentMeals)) {
      suggestions.push({
        type: 'health',
        message: "Défi : Ajoute 2 portions de légumes aujourd'hui pour débloquer un badge !",
        data: { vegetable_target: 2, reward: 'veggie_lover_badge' }
      });
    }

    return suggestions[Math.floor(Math.random() * suggestions.length)] || {
      type: 'general',
      message: "Continue comme ça ! Tu maintiens ta flamme depuis " + context.streak + " jours 🔥",
      data: {}
    };
  }

  // ============= SEASONAL EVENTS =============

  async checkSeasonalEvents(): Promise<ActiveEvent[]> {
    const today = new Date().toISOString().split('T')[0];

    const { data: events } = await supabase
      .from('seasonal_events')
      .select('*')
      .lte('start_date', today)
      .gte('end_date', today)
      .eq('is_active', true);

    const activeEvents = [];

  async updateEventProgress(eventId: string, challengeId: string): Promise<EventProgress> {
    const { data: progress } = await supabase
      .from('user_event_progress')
      .select('*')
      .eq('user_id', this.userId)
      .eq('event_id', eventId)
      .single();

    const updatedChallenges = {
      ...progress?.challenges_completed,
      [challengeId]: true
    };

    const totalCompleted = Object.values(updatedChallenges).filter(Boolean).length;

    await supabase
      .from('user_event_progress')
      .upsert({
        user_id: this.userId,
        event_id: eventId,
        challenges_completed: updatedChallenges,
        total_progress: totalCompleted
      });

    // Check for event completion
    const { data: event } = await supabase
      .from('seasonal_events')
      .select('*')
      .eq('event_id', eventId)
      .single();

    const allChallengesComplete = totalCompleted === Object.keys(event.challenges).length;

    if (allChallengesComplete) {
      await this.completeSeasonalEvent(eventId, event.rewards);
    }

    return {
      eventId,
      progress: totalCompleted,
      total: Object.keys(event.challenges).length,
      completed: allChallengesComplete
    };
  }

  // ============= NOTIFICATION SYSTEM =============

  async scheduleSmartNotifications(): Promise<void> {
    const { data: schedule } = await supabase
      .from('notification_schedule')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const notifications = [];

    // Morning notification
    if (schedule?.streak_reminders) {
      notifications.push({
        time: schedule.morning_time || '08:00',
        type: 'streak_reminder',
        message: await this.generateStreakMessage()
      });
    }

    // Lunch notification
    if (schedule?.quest_reminders) {
      notifications.push({
        time: schedule.lunch_time || '12:30',
        type: 'quest_update',
        message: await this.generateQuestMessage()
      });
    }

    // Evening notification
    if (schedule?.heart_refill_notifications) {
      notifications.push({
        time: schedule.evening_time || '20:00',
        type: 'daily_summary',
        message: await this.generateEveningMessage()
      });
    }

    // Schedule with notification service
    for (const notif of notifications) {
      await this.scheduleNotification(notif);
    }
  }

  private async generateStreakMessage(): Promise<string> {
    const streak = await this.getUserStreak();
    const messages = [
      `Jour ${streak} ! Ne perds pas ta flamme 🔥`,
      `${streak} jours de suite ! Continue comme ça 💪`,
      `Ta flamme brille depuis ${streak} jours ✨`,
      `Streak de ${streak} jours ! Tu es en feu 🔥`
    ];
    return messages[Math.floor(Math.random() * messages.length)];
  }

  // ============= XP & LEVEL SYSTEM =============

  async awardXP(amount: number, source: string): Promise<XPUpdate> {
    const { data: progress } = await supabase
      .from('gamification_progress')
      .select('*')
      .eq('user_id', this.userId)
      .single();

    const currentXP = progress?.experience_points || 0;
    const newXP = currentXP + amount;
    const currentLevel = progress?.level || 1;
    const newLevel = this.calculateLevel(newXP);

    await supabase
      .from('gamification_progress')
      .upsert({
        user_id: this.userId,
        experience_points: newXP,
        level: newLevel,
        total_xp_earned: (progress?.total_xp_earned || 0) + amount
      });

    const leveledUp = newLevel > currentLevel;

    if (leveledUp) {
      await this.handleLevelUp(newLevel);
    }

    analytics.track('xp_earned', {
      user_id: this.userId,
      amount,
      source,
      new_total: newXP,
      leveled_up: leveledUp
    });

    return {
      xpEarned: amount,
      totalXP: newXP,
      level: newLevel,
      leveledUp,
      nextLevelXP: this.getNextLevelXP(newLevel)
    };
  }

  private calculateLevel(xp: number): number {
    return Math.floor(Math.sqrt(xp / 50)) + 1;
  }

  private getNextLevelXP(level: number): number {
    return Math.pow(level, 2) * 50;
  }
}
