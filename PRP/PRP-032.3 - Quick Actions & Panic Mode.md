// services/panic-mode/panic-engine.ts
import { SupabaseClient } from '@supabase/supabase-js';

interface PanicContext {
  userId: string;
  timeAvailable: number;
  stressLevel: 1 | 2 | 3 | 4 | 5;
  familyMembers: number;
  currentLocation?: { lat: number; lng: number };
  inventory?: InventoryItem[];
  preferences?: UserPreferences;
}

interface PanicSolution {
  id: string;
  type: 'instant' | 'delivery' | 'prepared' | 'restaurant';
  title: string;
  description: string;
  timeRequired: number;
  estimatedCost?: number;
  difficulty: 'trivial' | 'easy' | 'medium';
  confidence: number; // 0-100
  steps?: string[];
  ingredients?: Ingredient[];
  restaurant?: RestaurantInfo;
  recipe?: Recipe;
}

export class PanicEngine {
  constructor(
    private supabase: SupabaseClient,
    private inventoryService: InventoryService,
    private deliveryService: DeliveryService,
    private recipeService: RecipeService
  ) {}

  async triggerPanic(context: PanicContext): Promise<PanicSolution[]> {
    const startTime = Date.now();
    
    try {
      // 1. Log l'événement
      await this.logPanicEvent(context);
      
      // 2. Vérifier le cache de solutions pré-calculées
      const cachedSolutions = await this.getCachedSolutions(context);
      if (cachedSolutions.length > 0) {
        return this.rankSolutions(cachedSolutions, context);
      }
      
      // 3. Générer des solutions en parallèle
      const [instant, delivery, prepared] = await Promise.all([
        this.generateInstantSolutions(context),
        this.generateDeliverySolutions(context),
        this.generatePreparedSolutions(context)
      ]);
      
      // 4. Combiner et ranker
      const allSolutions = [...instant, ...delivery, ...prepared];
      const rankedSolutions = this.rankSolutions(allSolutions, context);
      
      // 5. Mettre en cache pour la prochaine fois
      await this.cacheSolutions(rankedSolutions, context);
      
      // 6. Garantir max 3 solutions, temps < 30s
      const elapsedTime = Date.now() - startTime;
      if (elapsedTime > 30000) {
        console.error(`Panic resolution took ${elapsedTime}ms`);
        // Fallback sur solutions ultra-basiques
        return this.getEmergencySolutions();
      }
      
      return rankedSolutions.slice(0, 3);
      
    } catch (error) {
      console.error('Panic engine failed:', error);
      return this.getEmergencySolutions();
    }
  }

  private async generateInstantSolutions(context: PanicContext): Promise<PanicSolution[]> {
    const solutions: PanicSolution[] = [];
    
    // 1. Basé sur l'inventaire
    if (context.inventory && context.inventory.length > 0) {
      const quickRecipes = await this.recipeService.findQuickRecipesWithIngredients(
        context.inventory,
        context.timeAvailable || 15
      );
      
      solutions.push(...quickRecipes.map(recipe => ({
        id: `instant-${recipe.id}`,
        type: 'instant' as const,
        title: recipe.name,
        description: `Avec ce que vous avez: ${recipe.mainIngredients.join(', ')}`,
        timeRequired: recipe.prepTime + recipe.cookTime,
        difficulty: this.calculateDifficulty(recipe),
        confidence: this.calculateConfidence(recipe, context),
        recipe,
        steps: this.simplifySteps(recipe.steps)
      })));
    }
    
    // 2. Recettes d'urgence universelles
    solutions.push(
      {
        id: 'instant-pasta',
        type: 'instant',
        title: 'Pâtes Aglio e Olio',
        description: 'Pâtes, huile d\'olive, ail - 10 minutes chrono',
        timeRequired: 10,
        difficulty: 'trivial',
        confidence: 95,
        steps: [
          'Faire bouillir l\'eau',
          'Cuire les pâtes',
          'Faire revenir l\'ail dans l\'huile',
          'Mélanger et servir'
        ]
      },
      {
        id: 'instant-omelette',
        type: 'instant',
        title: 'Omelette garnie',
        description: 'Œufs + ce que vous trouvez',
        timeRequired: 8,
        difficulty: 'trivial',
        confidence: 90,
        steps: [
          'Battre les œufs',
          'Ajouter garniture disponible',
          'Cuire 3-4 minutes',
          'Plier et servir'
        ]
      }
    );
    
    return solutions;
  }

  private async generateDeliverySolutions(context: PanicContext): Promise<PanicSolution[]> {
    if (!context.currentLocation) return [];
    
    const nearbyOptions = await this.deliveryService.getNearbyDelivery(
      context.currentLocation,
      {
        maxDeliveryTime: 30,
        familySize: context.familyMembers,
        cuisinePreferences: context.preferences?.cuisines
      }
    );
    
    return nearbyOptions.map(option => ({
      id: `delivery-${option.id}`,
      type: 'delivery' as const,
      title: option.restaurantName,
      description: `${option.cuisine} - Livraison ${option.estimatedTime}min`,
      timeRequired: option.estimatedTime,
      estimatedCost: option.estimatedCost,
      difficulty: 'trivial' as const,
      confidence: 85,
      restaurant: option
    }));
  }

  private async generatePreparedSolutions(context: PanicContext): Promise<PanicSolution[]> {
    // Vérifier les plats préparés au congélateur
    const freezerMeals = await this.inventoryService.getFreezerMeals(context.userId);
    
    return freezerMeals.map(meal => ({
      id: `prepared-${meal.id}`,
      type: 'prepared' as const,
      title: meal.name,
      description: `Au congélateur - Décongélation ${meal.defrostTime}min`,
      timeRequired: meal.defrostTime + meal.reheatingTime,
      difficulty: 'trivial' as const,
      confidence: 100,
      steps: meal.reheatingInstructions
    }));
  }

  private rankSolutions(solutions: PanicSolution[], context: PanicContext): PanicSolution[] {
    return solutions.sort((a, b) => {
      // Scoring multi-critères
      const scoreA = this.calculateSolutionScore(a, context);
      const scoreB = this.calculateSolutionScore(b, context);
      return scoreB - scoreA;
    });
  }

  private calculateSolutionScore(solution: PanicSolution, context: PanicContext): number {
    let score = solution.confidence;
    
    // Bonus/Malus selon le contexte
    if (solution.timeRequired <= 10) score += 20;
    if (solution.difficulty === 'trivial') score += 15;
    if (context.stressLevel >= 4 && solution.type === 'instant') score += 25;
    if (solution.estimatedCost && solution.estimatedCost > 50) score -= 10;
    
    return Math.min(100, Math.max(0, score));
  }

  private async getEmergencySolutions(): Promise<PanicSolution[]> {
    // Solutions de dernier recours, toujours disponibles
    return [
      {
        id: 'emergency-cereals',
        type: 'instant',
        title: 'Céréales ou tartines',
        description: 'Le classique qui sauve',
        timeRequired: 2,
        difficulty: 'trivial',
        confidence: 100,
        steps: ['Sortir les céréales', 'Ajouter du lait', 'C\'est prêt!']
      },

        id: 'emergency-order',
        type: 'delivery',
        title: 'Commander une pizza',
        description: 'Appeler la pizzeria du coin',
        timeRequired: 30,
        difficulty: 'trivial',
        confidence: 95,
        steps: ['Téléphoner', 'Commander', 'Attendre']
      }
    ];
  }
}

// services/quick-actions/quick-actions-engine.ts

export class QuickActionsEngine {
  private actionHandlers: Map<string, ActionHandler>;

  constructor(
    private supabase: SupabaseClient,
    private planningService: PlanningService,
    private analyticsService: AnalyticsService
  ) {
    this.registerActionHandlers();
  }

  private registerActionHandlers() {
    this.actionHandlers = new Map([
      ['repeat_week', new RepeatWeekHandler(this.supabase)],
      ['survival_mode', new SurvivalModeHandler(this.supabase)],
      ['empty_fridge', new EmptyFridgeHandler(this.supabase)],
      ['reset_week', new ResetWeekHandler(this.supabase)],
      ['smart_suggest', new SmartSuggestHandler(this.supabase)],
      ['batch_cooking', new BatchCookingHandler(this.supabase)]
    ]);
  }

  async executeAction(
    actionType: string, 
    userId: string, 
    params?: any
  ): Promise<ActionResult> {
    const startTime = performance.now();
    
    try {
      const handler = this.actionHandlers.get(actionType);
      if (!handler) {
        throw new Error(`Unknown action type: ${actionType}`);
      }

      // Execute with timeout
      const result = await this.withTimeout(
        handler.execute(userId, params),
        5000 // 5 seconds max
      );

      // Log usage
      await this.logActionUsage(actionType, userId, startTime, true, result);
      
      return result;

    } catch (error) {
      await this.logActionUsage(actionType, userId, startTime, false, null, error);
      throw error;
    }
  }

  private async withTimeout<T>(promise: Promise<T>, ms: number): Promise<T> {
    const timeout = new Promise((_, reject) => 
      setTimeout(() => reject(new Error('Action timeout')), ms)
    );
    return Promise.race([promise, timeout]) as Promise<T>;
  }
}

// Exemple d'un handler spécifique
class RepeatWeekHandler implements ActionHandler {
  constructor(private supabase: SupabaseClient) {}

  async execute(userId: string, params?: any): Promise<ActionResult> {
    // 1. Récupérer le dernier plan complet
    const { data: lastPlan } = await this.supabase
      .from('weekly_meal_plans')
      .select(`
        *,
        meal_plan_entries(*)
      `)
      .eq('user_id', userId)
      .eq('status', 'completed')
      .order('week_start_date', { ascending: false })
      .limit(1)
      .single();

    if (!lastPlan) {
      throw new Error('Aucun plan précédent à répéter');
    }

    // 2. Cloner pour la semaine courante
    const newPlan = await this.clonePlanForCurrentWeek(lastPlan);

    // 3. Vérifier les disponibilités d'ingrédients
    const availability = await this.checkIngredientsAvailability(newPlan);

    return {
      success: true,
      data: newPlan,
      message: `Plan de la semaine du ${lastPlan.week_start_date} répété`,
      warnings: availability.missing.length > 0 
        ? `Ingrédients manquants: ${availability.missing.join(', ')}`
        : undefined
    };
  }
}

// hooks/useGestures.ts
import { useEffect } from 'react';
import { useSwipeable } from 'react-swipeable';
import { useLongPress } from 'use-long-press';

export const useQuickActionGestures = (onAction: (action: string) => void) => {
  // Swipe gestures
  const swipeHandlers = useSwipeable({
    onSwipedUp: () => onAction('panic_mode'),
    onSwipedDown: () => onAction('quick_plan'),
    onSwipedLeft: () => onAction('next_day'),
    onSwipedRight: () => onAction('previous_day'),
    preventDefaultTouchmoveEvent: true,
    trackMouse: false
  });

  // Long press for repeat week
  const longPressHandlers = useLongPress(
    () => onAction('repeat_week'),
    {
      isPreventDefault: true,
      delay: 500
    }
  );

  // Shake detection
  useEffect(() => {
    let lastX = 0, lastY = 0, lastZ = 0;
    let moveCounter = 0;

    const handleMotion = (event: DeviceMotionEvent) => {
      const { x, y, z } = event.accelerationIncludingGravity || {};
      
      if (x && y && z) {
        const deltaX = Math.abs(x - lastX);
        const deltaY = Math.abs(y - lastY);
        const deltaZ = Math.abs(z - lastZ);
        
        if (deltaX + deltaY + deltaZ > 15) {
          moveCounter++;
          
          if (moveCounter > 2) {
            onAction('panic_mode');
            moveCounter = 0;
          }
        }
        
        lastX = x;
        lastY = y;
        lastZ = z;
      }
    };

    if (window.DeviceMotionEvent) {
      window.addEventListener('devicemotion', handleMotion);
    }

    return () => {
      window.removeEventListener('devicemotion', handleMotion);
    };
  }, [onAction]);

  return {
    swipeHandlers,
    longPressHandlers
  };
};

  async generateInsights(userId: string): Promise<PanicInsights> {
    const { data: events } = await supabase
      .from('panic_events')
      .select('*')
      .eq('user_id', userId)
      .gte('created_at', new Date(Date.now() - 30 * 24 * 60 * 60 * 1000)); // Last 30 days

    return {
      panicFrequency: events.length / 30, // Per day
      peakPanicHour: this.findPeakHour(events),
      preferredSolution: this.findPreferredSolution(events),
      avgResolutionTime: this.calculateAvgTime(events),
      stressPattern: this.analyzeStressPattern(events),
      recommendations: this.generateRecommendations(events)
    };
  }

  private generateRecommendations(events: PanicEvent[]): string[] {
    const recommendations = [];
    
    // Si panic fréquent le même jour
    const dayPattern = this.findDayPattern(events);
    if (dayPattern) {
      recommendations.push(
        `Préparez spécialement le ${dayPattern} - c'est votre jour critique`
      );
    }

    // Si souvent vers 18h
    const timePattern = this.findTimePattern(events);
    if (timePattern === 18) {
      recommendations.push(
        'Activez les rappels à 17h30 pour éviter le stress de dernière minute'
      );
    }

    // Si préférence delivery
    if (this.findPreferredSolution(events) === 'delivery') {
      recommendations.push(
        'Planifiez 1-2 restos par semaine pour réduire la pression'
      );
    }

    return recommendations;
  }

SELECT 
  EXTRACT(HOUR FROM trigger_time) as hour,
  COUNT(*) as panic_count,
  AVG(time_to_resolution) as avg_resolution_time

WHERE created_at > NOW() - INTERVAL '7 days'
GROUP BY hour
ORDER BY hour;

SELECT 
  solution_selected,
  COUNT(*) as selection_count,
  AVG(CASE WHEN success THEN 1 ELSE 0 END) as success_rate

GROUP BY solution_selected
ORDER BY selection_count DESC;
