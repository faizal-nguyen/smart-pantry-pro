## 💻 Code Examples

### AppNavigation.tsx (Skelette)
```typescript
interface AppNavigationProps {
  children: React.ReactNode;
  user: User;
}

export function AppNavigation({ children, user }: AppNavigationProps) {
  const { isMobile, isTablet, isDesktop } = useBreakpoints();
  const { zones } = useResponsiveZones();
  
  return (
    <div className="min-h-screen bg-background">
      {isMobile && <MobileNavigation />}
      {isTablet && <TabletNavigation />}
      {isDesktop && <DesktopNavigation />}
      
      <main className={cn(
        "min-h-screen",
        !isMobile && "ml-64",
        isMobile && "pt-14 pb-20"
      )}>
        {children}
      </main>
    </div>
  );
}
```
