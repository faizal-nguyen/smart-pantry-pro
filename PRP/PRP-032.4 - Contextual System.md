# 📋 PRP-032.4 - Contextual System (Développement Complet)

## 🎯 Vision Produit

**Le problème** : 68% des plans de repas échouent car ils ignorent le contexte réel (météo, planning familial, saisons, promotions). Un plan parfait sur papier devient inadapté dans la vraie vie.

**Notre solution** : Un système contextuel intelligent qui adapte automatiquement les suggestions selon la météo, l'emploi du temps, les saisons et les opportunités économiques, créant des plans véritablement réalisables.

interface WeatherProvider {
  name: 'OpenWeatherMap' | 'WeatherAPI' | 'MeteoFrance';
  apiKey: string;
  baseUrl: string;
}

export class WeatherContextService {
  private providers: Map<string, WeatherProvider>;
  private cache: CacheService;

  constructor() {
    this.providers = new Map([
      ['primary', {
        name: 'OpenWeatherMap',
        apiKey: process.env.OPENWEATHER_API_KEY!,
        baseUrl: 'https://api.openweathermap.org/data/2.5'
      }],
      ['fallback', {
        name: 'WeatherAPI',
        apiKey: process.env.WEATHERAPI_KEY!,
        baseUrl: 'https://api.weatherapi.com/v1'
      }]
    ]);
  }

  async getWeatherContext(location: Coordinates): Promise<WeatherContext> {
    const cacheKey = `weather:${location.lat}:${location.lng}`;
    
    // Check cache first
    const cached = await this.cache.get(cacheKey);
    if (cached) return cached;

    try {
      // Try primary provider
      const weather = await this.fetchWeatherData('primary', location);
      const context = this.processWeatherData(weather);
      
      // Cache for 1 hour
      await this.cache.set(cacheKey, context, 3600);
      
      return context;
    } catch (error) {
      console.error('Primary weather provider failed:', error);
      // Fallback to secondary
      return this.fetchWeatherData('fallback', location);
    }
  }

  private async fetchWeatherData(
    provider: string, 
    location: Coordinates
  ): Promise<RawWeatherData> {
    const config = this.providers.get(provider)!;
    
    if (config.name === 'OpenWeatherMap') {
      const response = await fetch(
        `${config.baseUrl}/forecast?lat=${location.lat}&lon=${location.lng}&appid=${config.apiKey}&units=metric`
      );
      return response.json();
    }
    
    // Implement other providers...
    throw new Error('Provider not implemented');
  }

  private processWeatherData(raw: RawWeatherData): WeatherContext {
    const forecast = raw.list.map(item => ({
      date: new Date(item.dt * 1000),
      temp: item.main.temp,
      feels_like: item.main.feels_like,
      humidity: item.main.humidity,
      weather: item.weather[0].main,
      description: item.weather[0].description,
      rain: item.rain?.['3h'] || 0,
      wind: item.wind.speed
    }));

    // Analyze patterns
    const weeklyAnalysis = {
      avgTemp: this.calculateAverage(forecast.map(f => f.temp)),
      tempTrend: this.detectTrend(forecast.map(f => f.temp)),
      rainyDays: forecast.filter(f => f.rain > 0).length,
      extremeWeather: this.detectExtremes(forecast),
      recommendations: this.generateWeatherRecommendations(forecast)
    };

    return {
      current: forecast[0],
      forecast: forecast.slice(0, 7 * 8), // 7 days, 3-hour intervals
      analysis: weeklyAnalysis,
      impact: this.calculateWeatherImpact(weeklyAnalysis)
    };
  }

  private generateWeatherRecommendations(forecast: WeatherData[]): MealRecommendation[] {
    const recommendations: MealRecommendation[] = [];
    
    forecast.forEach((weather, index) => {
      const day = Math.floor(index / 8);
      
      // Très chaud (>30°C)
      if (weather.temp > 30) {
        recommendations.push({
          day,
          type: 'weather_hot',
          suggestion: 'salad',
          reason: `Température élevée prévue (${weather.temp}°C)`,
          alternatives: ['gazpacho', 'cold_pasta', 'poke_bowl']
        });
      }
      
      // Très froid (<5°C)
      if (weather.temp < 5) {
        recommendations.push({
          day,
          type: 'weather_cold',
          suggestion: 'soup',
          reason: `Température basse prévue (${weather.temp}°C)`,
          alternatives: ['pot_au_feu', 'gratin', 'raclette']
        });
      }
      
      // Pluvieux
      if (weather.rain > 5) {
        recommendations.push({
          day,
          type: 'weather_rain',
          suggestion: 'comfort_food',
          reason: 'Jour de pluie prévu',
          alternatives: ['pasta', 'risotto', 'curry']
        });
      }
    });
    
    return recommendations;
  }

  private calculateWeatherImpact(analysis: WeeklyWeatherAnalysis): WeatherImpact {
    return {
      mealTypeImpact: {
        hot_meals: analysis.avgTemp < 15 ? 1.5 : 0.8,
        cold_meals: analysis.avgTemp > 25 ? 1.5 : 0.8,
        comfort_food: analysis.rainyDays > 3 ? 1.3 : 1.0,
        bbq: analysis.avgTemp > 20 && analysis.rainyDays < 2 ? 1.4 : 0.6
      },
      cookingMethodImpact: {
        oven: analysis.avgTemp > 28 ? 0.7 : 1.0, // Éviter le four si chaud
        stovetop: 1.0,
        no_cook: analysis.avgTemp > 30 ? 1.5 : 0.9,
        slow_cooker: analysis.avgTemp < 10 ? 1.3 : 1.0
      }
    };
  }
}

export class CalendarContextService {
  private googleCalendar: GoogleCalendarClient;
  private outlookCalendar: OutlookCalendarClient;
  
  async getCalendarContext(userId: string, weekStart: Date): Promise<CalendarContext> {
    const preferences = await this.getUserCalendarPreferences(userId);
    const events: CalendarEvent[] = [];
    
    // Fetch from connected calendars
    if (preferences.google_calendar_connected) {
      const googleEvents = await this.fetchGoogleEvents(
        preferences.google_calendar_token,
        weekStart
      );
      events.push(...googleEvents);
    }
    
    if (preferences.outlook_calendar_connected) {
      const outlookEvents = await this.fetchOutlookEvents(
        preferences.outlook_calendar_token,
        weekStart
      );
      events.push(...outlookEvents);
    }
    
    // Process events for meal planning
    return this.processCalendarEvents(events, weekStart);
  }

  private async fetchGoogleEvents(
    token: string, 
    weekStart: Date
  ): Promise<CalendarEvent[]> {
    const calendar = google.calendar({ version: 'v3', auth: token });
    const weekEnd = addDays(weekStart, 7);
    
    const response = await calendar.events.list({
      calendarId: 'primary',
      timeMin: weekStart.toISOString(),
      timeMax: weekEnd.toISOString(),
      singleEvents: true,
      orderBy: 'startTime'
    });
    
    return response.data.items?.map(event => ({
      id: event.id!,
      title: event.summary || 'Sans titre',
      start: new Date(event.start?.dateTime || event.start?.date!),
      end: new Date(event.end?.dateTime || event.end?.date!),
      allDay: !event.start?.dateTime,
      location: event.location,
      attendees: event.attendees?.length || 1,
      type: this.categorizeEvent(event.summary || ''),
      impact: this.calculateEventImpact(event)
    })) || [];
  }

  private categorizeEvent(title: string): EventCategory {
    const lowerTitle = title.toLowerCase();
    
    if (lowerTitle.includes('anniversaire') || lowerTitle.includes('birthday')) {
      return 'birthday';
    }
    if (lowerTitle.includes('meeting') || lowerTitle.includes('réunion')) {
      return 'work_meeting';
    }
    if (lowerTitle.includes('sport') || lowerTitle.includes('gym')) {
      return 'sport';
    }
    if (lowerTitle.includes('voyage') || lowerTitle.includes('travel')) {
      return 'travel';
    }
    if (lowerTitle.includes('dîner') || lowerTitle.includes('dinner')) {
      return 'dinner_event';
    }
    
    return 'other';
  }

  private processCalendarEvents(
    events: CalendarEvent[], 
    weekStart: Date
  ): CalendarContext {
    const daySchedules: DaySchedule[] = [];
    
    for (let i = 0; i < 7; i++) {
      const currentDay = addDays(weekStart, i);
      const dayEvents = events.filter(e => 
        isSameDay(e.start, currentDay)
      );
      
      const busyScore = this.calculateBusyScore(dayEvents);
      const mealTimeAvailable = this.calculateMealTimeAvailable(dayEvents);
      const recommendations = this.generateScheduleRecommendations(dayEvents);
      
      daySchedules.push({
        date: currentDay,
        events: dayEvents,
        busyScore, // 0-10
        mealTimeAvailable,
        recommendations,
        suggestedMealComplexity: busyScore > 7 ? 'simple' : 
                                 busyScore > 4 ? 'medium' : 'complex'
      });
    }
    
    return {
      weekSchedule: daySchedules,
      specialOccasions: events.filter(e => 
        ['birthday', 'dinner_event'].includes(e.type)
      ),
      overallBusyScore: this.calculateAverage(daySchedules.map(d => d.busyScore)),
      recommendations: this.consolidateRecommendations(daySchedules)
    };
  }

  private calculateBusyScore(events: CalendarEvent[]): number {
    let score = 0;
    
    // Base sur le nombre d'événements
    score += Math.min(events.length * 2, 6);
    
    // Ajout selon les types d'événements
    events.forEach(event => {
      if (event.type === 'work_meeting') score += 1;
      if (event.type === 'travel') score += 3;
      if (event.type === 'sport') score += 1;
      
      // Événements en soirée = plus d'impact
      const hour = event.start.getHours();
      if (hour >= 17 && hour <= 20) score += 2;
    });
    
    return Math.min(score, 10);
  }

  private calculateMealTimeAvailable(events: CalendarEvent[]): MealTimeSlots {
    const lunchTime = { start: 12, end: 14 };
    const dinnerTime = { start: 19, end: 21 };
    
    const lunchConflicts = events.filter(e => {
      const hour = e.start.getHours();
      return hour >= lunchTime.start && hour < lunchTime.end;
    });
    
    const dinnerConflicts = events.filter(e => {
      const hour = e.start.getHours();
      return hour >= dinnerTime.start && hour < dinnerTime.end;
    });
    
    return {
      lunch: {
        available: lunchConflicts.length === 0,
        timeWindow: lunchConflicts.length > 0 ? 30 : 60,
        suggestion: lunchConflicts.length > 0 ? 'quick_lunch' : 'normal_lunch'
      },
      dinner: {
        available: dinnerConflicts.length === 0,
        timeWindow: dinnerConflicts.length > 0 ? 45 : 90,
        suggestion: dinnerConflicts.length > 0 ? 'prepare_ahead' : 'normal_dinner'
      }
    };
  }
}

interface SeasonalIngredient {
  id: string;
  name: string;
  category: 'fruit' | 'vegetable' | 'fish' | 'meat';
  peakMonths: number[]; // 1-12
  availability: {
    [month: number]: {
      inSeason: boolean;
      quality: 'poor' | 'good' | 'excellent';
      priceIndex: number; // 0.5 = 50% moins cher, 1.5 = 50% plus cher
    };
  };
  origin: 'local' | 'national' | 'imported';
}

export class SeasonalityEngine {
  private seasonalDB: Map<string, SeasonalIngredient>;
  
  constructor() {
    this.initializeSeasonalDatabase();
  }

  private initializeSeasonalDatabase() {
    // Base de données des produits de saison (France)
    this.seasonalDB = new Map([
      ['tomato', {
        id: 'tomato',
        name: 'Tomate',
        category: 'vegetable',

          5: { inSeason: true, quality: 'excellent', priceIndex: 0.7 },
          6: { inSeason: true, quality: 'excellent', priceIndex: 0.6 },
          7: { inSeason: true, quality: 'good', priceIndex: 1.0 },
          8: { inSeason: false, quality: 'poor', priceIndex: 1.5 },
          9: { inSeason: false, quality: 'poor', priceIndex: 2.0 },
          10: { inSeason: false, quality: 'poor', priceIndex: 2.5 },
          11: { inSeason: false, quality: 'poor', priceIndex: 3.0 },
          12: { inSeason: false, quality: 'poor', priceIndex: 3.0 }
        },
        origin: 'local'
      }],
      // ... Ajouter tous les produits
    ]);
  }

  async getSeasonalContext(date: Date = new Date()): Promise<SeasonalContext> {
    const month = date.getMonth() + 1;
    const season = this.getCurrentSeason(month);
    
    // Produits du mois
    const inSeasonProducts = Array.from(this.seasonalDB.values())
      .filter(product => product.availability[month].inSeason)
      .sort((a, b) => {
        // Prioriser par qualité puis prix
        const qualityScore = { excellent: 3, good: 2, poor: 1 };
        const aScore = qualityScore[a.availability[month].quality] / a.availability[month].priceIndex;
        const bScore = qualityScore[b.availability[month].quality] / b.availability[month].priceIndex;
        return bScore - aScore;
      });

    // Recettes de saison
    const seasonalRecipes = await this.getSeasonalRecipes(inSeasonProducts, season);
    
    // Produits bientôt disponibles
    const upcomingProducts = this.getUpcomingProducts(month);
    
    // Produits en fin de saison (à consommer)
    const endingProducts = this.getEndingProducts(month);

    return {
      currentSeason: season,
      month: month,
      inSeasonProducts: inSeasonProducts.map(p => ({
        ...p,
        score: this.calculateSeasonalScore(p, month)
      })),
      seasonalRecipes,
      upcomingProducts,
      endingProducts,
      recommendations: this.generateSeasonalRecommendations(
        inSeasonProducts,
        season
      )
    };
  }

  private getCurrentSeason(month: number): Season {
    if (month >= 3 && month <= 5) return 'spring';
    if (month >= 6 && month <= 8) return 'summer';
    if (month >= 9 && month <= 11) return 'autumn';
    return 'winter';
  }

  private async getSeasonalRecipes(
    products: SeasonalIngredient[], 
    season: Season
  ): Promise<Recipe[]> {
    // Requête à la base de recettes avec boost saisonnier
    const { data: recipes } = await supabase
      .from('recipes_catalog')
      .select('*')
      .contains('main_ingredients', products.map(p => p.name))
      .eq('season_tag', season)
      .limit(20);

    return recipes || [];
  }

  private calculateSeasonalScore(
    product: SeasonalIngredient, 
    month: number
  ): number {
    const availability = product.availability[month];
    let score = 0;
    
    // Score basé sur la saison
    if (availability.inSeason) score += 40;
    
    // Score basé sur la qualité
    switch (availability.quality) {
      case 'excellent': score += 30; break;
      case 'good': score += 20; break;
      case 'poor': score += 5; break;
    }
    
    // Score basé sur le prix
    if (availability.priceIndex < 0.8) score += 30;
    else if (availability.priceIndex < 1.2) score += 15;
    
    // Bonus local
    if (product.origin === 'local') score += 10;
    
    return Math.min(100, score);
  }

  private generateSeasonalRecommendations(
    products: SeasonalIngredient[],
    season: Season
  ): SeasonalRecommendation[] {
    const recommendations: SeasonalRecommendation[] = [];
    
    // Recommandations par saison
    const seasonalThemes = {
      spring: {
        themes: ['detox', 'fresh', 'green'],
        recipes: ['salades printanières', 'asperges grillées', 'risotto aux petits pois']
      },
      summer: {
        themes: ['bbq', 'cold', 'light'],
        recipes: ['gazpacho', 'salades composées', 'grillades', 'tartes aux fruits']
      },
      autumn: {
        themes: ['comfort', 'roasted', 'harvest'],
        recipes: ['soupes de potiron', 'gratins', 'tartes aux pommes', 'plats mijotés']
      },
      winter: {
        themes: ['warming', 'stew', 'preserved'],
        recipes: ['pot-au-feu', 'raclette', 'choucroute', 'soupes épaisses']
      }
    };
    
    const currentTheme = seasonalThemes[season];
    
    // Produits stars du moment
    const topProducts = products.slice(0, 5);
    topProducts.forEach(product => {
      recommendations.push({
        type: 'ingredient_spotlight',
        title: `${product.name} en pleine saison!`,
        description: `Profitez des ${product.name}s, excellente qualité et prix bas`,
        recipes: [], // À remplir avec les recettes correspondantes
        priority: 'high'
      });
    });
    
    // Thèmes saisonniers
    currentTheme.recipes.forEach(recipe => {
      recommendations.push({
        type: 'seasonal_theme',
        title: recipe,
        description: `Parfait pour la saison`,
        priority: 'medium'
      });
    });
    
    return recommendations;
  }
}

// services/context/promotions-service.ts

interface StorePromotion {
  storeId: string;
  storeName: string;
  distance: number; // km
  promotion: {
    id: string;
    product: string;
    originalPrice: number;
    discountedPrice: number;
    discountPercent: number;
    validUntil: Date;
    quantity?: string; // "2 pour 1", etc.
  };
}

export class PromotionsContextService {
  private storeAPIs: Map<string, StoreAPIClient>;
  
  constructor() {
    this.storeAPIs = new Map([
      ['carrefour', new CarrefourAPIClient()],
      ['leclerc', new LeclercAPIClient()],
      ['intermarche', new IntermarcheAPIClient()],
      ['lidl', new LidlAPIClient()]
    ]);
  }

  async getPromotionsContext(
    userLocation: Coordinates,
    preferences: UserPreferences
  ): Promise<PromotionsContext> {
    // 1. Identifier les magasins proches
    const nearbyStores = await this.getNearbyStores(
      userLocation, 
      preferences.preferred_stores || []
    );
    
    // 2. Récupérer les promotions en parallèle
    const promotionsPromises = nearbyStores.map(store => 
      this.getStorePromotions(store)
    );
    
    const allPromotions = await Promise.allSettled(promotionsPromises);
    
    // 3. Filtrer et scorer les promotions
    const relevantPromotions = this.filterRelevantPromotions(
      allPromotions
        .filter(r => r.status === 'fulfilled')
        .flatMap(r => (r as PromiseFulfilledResult<StorePromotion[]>).value),
      preferences
    );
    
    // 4. Grouper par catégorie et calculer l'impact
    return {
      promotions: relevantPromotions,
      byCategory: this.groupByCategory(relevantPromotions),
      totalSavingsPotential: this.calculateSavingsPotential(relevantPromotions),
      recommendations: this.generatePromotionRecommendations(relevantPromotions),
      expiringToday: this.getExpiringPromotions(relevantPromotions, 1),
      bestDeals: this.getBestDeals(relevantPromotions, 10)
    };
  }

  private async getNearbyStores(
    location: Coordinates,
    preferredStores: string[]
  ): Promise<Store[]> {
    // Utiliser une API de géolocalisation ou base de données locale
    const { data: stores } = await supabase
      .from('stores')
      .select('*')
      .lt('distance', 10) // Dans un rayon de 10km
      .order('distance');
    
    // Prioriser les magasins préférés
    return stores?.sort((a, b) => {
      const aPreferred = preferredStores.includes(a.id);
      const bPreferred = preferredStores.includes(b.id);
      if (aPreferred && !bPreferred) return -1;
      if (!aPreferred && bPreferred) return 1;
      return a.distance - b.distance;
    }) || [];
  }

  private async getStorePromotions(store: Store): Promise<StorePromotion[]> {
    const client = this.storeAPIs.get(store.chain);
    if (!client) return [];
    
    try {
      const promotions = await client.getPromotions(store.id);
      return promotions.map(promo => ({
        storeId: store.id,
        storeName: store.name,
        distance: store.distance,
        promotion: promo
      }));
    } catch (error) {
      console.error(`Failed to get promotions for ${store.name}:`, error);
      return [];
    }
  }

  private filterRelevantPromotions(
    promotions: StorePromotion[],
    preferences: UserPreferences
  ): StorePromotion[] {
    return promotions
      .filter(p => {
        // Filtrer par budget si spécifié
        if (preferences.max_price_per_item && 
            p.promotion.discountedPrice > preferences.max_price_per_item) {
          return false;
        }
        
        // Filtrer par catégories blacklistées
        if (preferences.excluded_categories?.includes(
          this.getProductCategory(p.promotion.product)
        )) {
          return false;
        }
        
        return true;
      })
      .sort((a, b) => {
        // Trier par économies absolues
        const savingsA = a.promotion.originalPrice - a.promotion.discountedPrice;
        const savingsB = b.promotion.originalPrice - b.promotion.discountedPrice;
        return savingsB - savingsA;
      });
  }

  private generatePromotionRecommendations(
    promotions: StorePromotion[]
  ): PromotionRecommendation[] {
    const recommendations: PromotionRecommendation[] = [];
    
    // Identifier les patterns
    const meatPromotions = promotions.filter(p => 
      this.getProductCategory(p.promotion.product) === 'meat'
    );
    
    if (meatPromotions.length > 3) {
      recommendations.push({
        type: 'bulk_opportunity',
        title: 'Opportunité viande',
        description: 'Plusieurs promotions sur la viande - pensez congélation!',
        products: meatPromotions.slice(0, 3),
        savingsPotential: this.calculateSavingsPotential(meatPromotions)
      });
    }
    
    // Promotions expiring soon
    const expiringSoon = this.getExpiringPromotions(promotions, 2);
    if (expiringSoon.length > 0) {
      recommendations.push({
        type: 'urgency',
        title: 'Derniers jours!',
        description: `${expiringSoon.length} promotions se terminent bientôt`,
        products: expiringSoon,
        priority: 'high'
      });
    }
    
    // Combo deals
    const comboOpportunities = this.findComboOpportunities(promotions);
    comboOpportunities.forEach(combo => {
      recommendations.push({
        type: 'combo',
        title: combo.recipeName,
        description: `Tous les ingrédients en promo pour ${combo.recipeName}`,
        products: combo.products,
        totalSavings: combo.savings
      });
    });
    
    return recommendations;
  }

  private findComboOpportunities(
    promotions: StorePromotion[]
  ): ComboOpportunity[] {
    // Trouver des recettes où plusieurs ingrédients sont en promo
    const combos: ComboOpportunity[] = [];
    
    // Exemple: Bolognaise
    const bologneseIngredients = ['viande hachée', 'tomates', 'pâtes', 'oignon'];
    const bolognesePromos = promotions.filter(p => 
      bologneseIngredients.some(ing => 
        p.promotion.product.toLowerCase().includes(ing)
      )
    );
    
    if (bolognesePromos.length >= 3) {
      combos.push({
        recipeName: 'Sauce Bolognaise',
        products: bolognesePromos,
        savings: this.calculateSavingsPotential(bolognesePromos)
      });
    }
    
    return combos;
  }
}

// services/context/context-adapter.ts

export class ContextAdapter {
  private weatherService: WeatherContextService;
  private calendarService: CalendarContextService;
  private seasonalityEngine: SeasonalityEngine;
  private promotionsService: PromotionsContextService;
  
  constructor() {
    this.weatherService = new WeatherContextService();
    this.calendarService = new CalendarContextService();
    this.seasonalityEngine = new SeasonalityEngine();
    this.promotionsService = new PromotionsContextService();
  }

  async adaptMealPlan(
    basePlan: MealPlan,
    userId: string,
    preferences: UserContextPreferences
  ): Promise<AdaptedMealPlan> {
    const startTime = performance.now();
    
    // 1. Gather all context data in parallel
    const [weather, calendar, seasonal, promotions] = await Promise.all([
      preferences.weather_adaptation 
        ? this.weatherService.getWeatherContext(preferences.home_location)
        : null,
      preferences.calendar_sync
        ? this.calendarService.getCalendarContext(userId, basePlan.weekStart)
        : null,
      preferences.seasonal_preferences
        ? this.seasonalityEngine.getSeasonalContext()
        : null,
      preferences.price_optimization
        ? this.promotionsService.getPromotionsContext(
            preferences.home_location,
            preferences
          )
        : null
    ]);
    
    // 2. Apply adaptations in order of priority
    let adaptedPlan = { ...basePlan };
    const adaptationLog: AdaptationLog[] = [];
    
    // Calendar has highest priority (can't cook if not home)
    if (calendar) {
      const calendarAdaptation = this.adaptToSchedule(adaptedPlan, calendar);
      adaptedPlan = calendarAdaptation.plan;
      adaptationLog.push(...calendarAdaptation.changes);
    }
    
    // Weather affects meal type preferences
    if (weather) {
      const weatherAdaptation = this.adaptToWeather(adaptedPlan, weather);
      adaptedPlan = weatherAdaptation.plan;
      adaptationLog.push(...weatherAdaptation.changes);
    }
    
    // Seasonality for ingredient choices
    if (seasonal) {
      const seasonalAdaptation = this.adaptToSeasons(adaptedPlan, seasonal);
      adaptedPlan = seasonalAdaptation.plan;
      adaptationLog.push(...seasonalAdaptation.changes);
    }
    
    // Promotions for cost optimization
    if (promotions) {
      const promotionAdaptation = this.adaptToPromotions(adaptedPlan, promotions);
      adaptedPlan = promotionAdaptation.plan;
      adaptationLog.push(...promotionAdaptation.changes);
    }
    
    // 3. Calculate confidence and impact
    const confidence = this.calculateConfidence(adaptationLog);
    const impact = this.calculateImpact(basePlan, adaptedPlan);
    
    // 4. Log performance
    const adaptationTime = performance.now() - startTime;
    await this.logAdaptation(userId, basePlan.id, adaptationLog, adaptationTime);
    
    return {
      originalPlan: basePlan,
      adaptedPlan,
      adaptations: adaptationLog,
      confidence,
      impact,
      executionTime: adaptationTime,
      context: { weather, calendar, seasonal, promotions }
    };
  }

  private adaptToWeather(
    plan: MealPlan, 
    weather: WeatherContext
  ): AdaptationResult {
    const changes: AdaptationLog[] = [];
    const adaptedMeals = [...plan.meals];
    
    weather.forecast.forEach((dayWeather, dayIndex) => {
      if (dayIndex >= adaptedMeals.length) return;
      
      const meal = adaptedMeals[dayIndex];
      
      // Très chaud -> privilégier les plats froids
      if (dayWeather.temp > 30) {
        if (meal.recipe.tags?.includes('hot_dish')) {
          const coldAlternative = this.findColdAlternative(meal.recipe);
          if (coldAlternative) {
            adaptedMeals[dayIndex] = {
              ...meal,
              recipe: coldAlternative,
              adaptationReason: `Température élevée (${dayWeather.temp}°C)`
            };
            
            changes.push({
              type: 'weather',
              day: dayIndex,
              original: meal.recipe.id,
              adapted: coldAlternative.id,
              reason: `Température ${dayWeather.temp}°C - plat froid recommandé`,
              confidence: 0.85
            });
          }
        }
      }
      
      // Très froid -> privilégier les plats réconfortants
      if (dayWeather.temp < 5) {
        if (meal.recipe.tags?.includes('cold_dish')) {
          const hotAlternative = this.findHotAlternative(meal.recipe);
          if (hotAlternative) {
            adaptedMeals[dayIndex] = {
              ...meal,
              recipe: hotAlternative,
              adaptationReason: `Température basse (${dayWeather.temp}°C)`
            };
            
            changes.push({
              type: 'weather',
              day: dayIndex,
              original: meal.recipe.id,
              adapted: hotAlternative.id,
              reason: `Température ${dayWeather.temp}°C - plat chaud recommandé`,
              confidence: 0.85
            });
          }
        }
      }
      
      // Pluie -> comfort food
      if (dayWeather.rain > 5) {
        const comfortAlternative = this.findComfortFood(meal.recipe);
        if (comfortAlternative && Math.random() > 0.5) { // 50% chance
          adaptedMeals[dayIndex] = {
            ...meal,
            recipe: comfortAlternative,
            adaptationReason: 'Jour de pluie'
          };
          
          changes.push({
            type: 'weather',
            day: dayIndex,
            original: meal.recipe.id,
            adapted: comfortAlternative.id,
            reason: 'Jour pluvieux - comfort food suggéré',
            confidence: 0.7
          });
        }
      }
    });
    
    return {
      plan: { ...plan, meals: adaptedMeals },
      changes
    };
  }

  private adaptToSchedule(
    plan: MealPlan,
    calendar: CalendarContext
  ): AdaptationResult {
    const changes: AdaptationLog[] = [];
    const adaptedMeals = [...plan.meals];
    
    calendar.weekSchedule.forEach((daySchedule, dayIndex) => {
      if (dayIndex >= adaptedMeals.length) return;
      
      const meal = adaptedMeals[dayIndex];
      
      // Jour très chargé -> repas rapide
      if (daySchedule.busyScore > 7) {
        const quickAlternative = this.findQuickAlternative(
          meal.recipe,
          daySchedule.mealTimeAvailable.dinner.timeWindow
        );
        
        if (quickAlternative) {
          adaptedMeals[dayIndex] = {
            ...meal,
            recipe: quickAlternative,
            adaptationReason: 'Journée chargée',
            prepTip: 'Préparer la veille si possible'
          };
          
          changes.push({
            type: 'schedule',
            day: dayIndex,
            original: meal.recipe.id,
            adapted: quickAlternative.id,
            reason: `Journée chargée (score ${daySchedule.busyScore}/10)`,
            confidence: 0.9
          });
        }
      }
      
      // Occasion spéciale
      const specialEvent = daySchedule.events.find(e => 
        e.type === 'birthday' || e.type === 'dinner_event'
      );
      
      if (specialEvent) {
        const specialMeal = this.findSpecialOccasionMeal(
          specialEvent.type,
          specialEvent.attendees || 2
        );
        
        if (specialMeal) {
          adaptedMeals[dayIndex] = {
            ...meal,
            recipe: specialMeal,
            servings: specialEvent.attendees || meal.servings,
            adaptationReason: `${specialEvent.title}`
          };
          
          changes.push({
            type: 'special_occasion',
            day: dayIndex,
            original: meal.recipe.id,
            adapted: specialMeal.id,
            reason: `Occasion spéciale: ${specialEvent.title}`,
            confidence: 0.95
          });
        }
      }
    });
    
    return {
      plan: { ...plan, meals: adaptedMeals },
      changes
    };
  }

  private adaptToSeasons(
    plan: MealPlan,
    seasonal: SeasonalContext
  ): AdaptationResult {
    const changes: AdaptationLog[] = [];
    const adaptedMeals = [...plan.meals];
    
    // Boost recipes with seasonal ingredients
    adaptedMeals.forEach((meal, index) => {
      const seasonalScore = this.calculateRecipeSeasonalScore(
        meal.recipe,
        seasonal.inSeasonProducts
      );
      
      if (seasonalScore < 50) {
        // Try to find a more seasonal alternative
        const seasonalAlternative = this.findSeasonalAlternative(
          meal.recipe,
          seasonal
        );
        
        if (seasonalAlternative && seasonalAlternative.seasonalScore > seasonalScore + 20) {
          adaptedMeals[index] = {
            ...meal,
            recipe: seasonalAlternative,
            adaptationReason: 'Produits de saison'
          };
          
          changes.push({
            type: 'seasonal',
            day: index,
            original: meal.recipe.id,
            adapted: seasonalAlternative.id,
            reason: `Amélioration saisonnière (+${Math.round(seasonalAlternative.seasonalScore - seasonalScore)} points)`,
            confidence: 0.75
          });
        }
      }
    });
    
    return {
      plan: { ...plan, meals: adaptedMeals },
      changes
    };
  }

  private adaptToPromotions(
    plan: MealPlan,
    promotions: PromotionsContext
  ): AdaptationResult {
    const changes: AdaptationLog[] = [];
    const adaptedMeals = [...plan.meals];
    
    // Find recipes that can benefit from promotions
    promotions.recommendations.forEach(recommendation => {
      if (recommendation.type === 'combo') {
        // Find a good day to use this combo
        const bestDayIndex = this.findBestDayForRecipe(
          recommendation.recipeName,
          adaptedMeals
        );
        
        if (bestDayIndex !== -1) {
          const promoRecipe = this.createRecipeFromPromotion(recommendation);
          
          adaptedMeals[bestDayIndex] = {
            ...adaptedMeals[bestDayIndex],
            recipe: promoRecipe,
            adaptationReason: `Économie de ${recommendation.totalSavings}€`,
            shoppingTip: `Acheter chez ${recommendation.products[0].storeName}`
          };
          
          changes.push({
            type: 'promotion',
            day: bestDayIndex,
            original: adaptedMeals[bestDayIndex].recipe.id,
            adapted: promoRecipe.id,
            reason: `Promotion: ${recommendation.title} (-${recommendation.totalSavings}€)`,
            confidence: 0.8,
            savings: recommendation.totalSavings
          });
        }
      }
    });
    
    return {
      plan: { ...plan, meals: adaptedMeals },
      changes
    };
  }

  private calculateConfidence(adaptations: AdaptationLog[]): number {
    if (adaptations.length === 0) return 1.0;
    
    const avgConfidence = adaptations.reduce((sum, a) => sum + a.confidence, 0) / adaptations.length;
    return Math.round(avgConfidence * 100) / 100;
  }

  private calculateImpact(original: MealPlan, adapted: MealPlan): PlanImpact {
    const changes = adapted.meals.filter((meal, i) => 
      meal.recipe.id !== original.meals[i]?.recipe.id
    ).length;
    
    const costDiff = this.calculateCostDifference(original, adapted);
    const timeDiff = this.calculateTimeDifference(original, adapted);
    const nutritionDiff = this.calculateNutritionDifference(original, adapted);
    
    return {
      changesCount: changes,
      changePercent: (changes / original.meals.length) * 100,
      costImpact: costDiff,
      timeImpact: timeDiff,
      nutritionImpact: nutritionDiff,
      overallScore: this.calculateOverallImpactScore(costDiff, timeDiff, nutritionDiff)
    };
  }
}

ORDER BY count DESC;
