              </p>
            </div>
          </div>
          
          {/* Mode Switcher */}
          <ModeToggle value={mode} onChange={setMode} />
        </div>
        
        {/* Quick Actions */}
        <div className="flex gap-2 mt-3 overflow-x-auto">
          <QuickAction

      {/* Messages */}

      <div className="border-t bg-white p-4">
        {mode === 'text' ? (
