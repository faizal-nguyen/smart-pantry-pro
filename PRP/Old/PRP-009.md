// src/services/voice/enhancedVoiceService.ts
import { toast } from 'sonner';

interface VoiceConfig {
  language: string;
  continuous: boolean;
  interimResults: boolean;
  maxAlternatives: number;
}

interface ParsedVoiceInput {
  quantity?: number;
  unit?: string;
  product: string;
  action: 'add' | 'remove' | 'search' | 'unknown';
  confidence: number;
}

export class EnhancedVoiceService {
  private recognition: SpeechRecognition | null = null;
  private isSupported = false;
  private isListening = false;
  private retryCount = 0;
  private maxRetries = 3;

  // Configuration optimisée pour le français
  private config: VoiceConfig = {
    language: 'fr-FR',
    continuous: false,
    interimResults: true,
    maxAlternatives: 3
  };

  // Regex patterns améliorés pour le français
  private patterns = {
    // Quantités avec unités
    quantityWithUnit: /(?:j'ai|ajoute|rajoute|met|mets)\s+(\d+(?:[.,]\d+)?)\s*(kg|g|grammes?|kilos?|litres?|l|ml|millilitres?|unités?|pièces?|boîtes?|paquets?|sachets?)\s+(?:de\s+)?(.+)/gi,
    
    // Quantités simples
    simpleQuantity: /(?:j'ai|ajoute|rajoute|met|mets)\s+(\d+(?:[.,]\d+)?)\s+(.+)/gi,
    
    // Produits sans quantité
    productOnly: /(?:j'ai|ajoute|rajoute|met|mets)\s+(?:du|de la|des|de l'|un|une|le|la|les)?\s*(.+)/gi,
    
    // Actions de suppression
    remove: /(?:enlève|supprime|retire|fini|terminé|plus de)\s+(.+)/gi,
    
    // Actions de recherche
    search: /(?:qu'est-ce que|que puis-je|recette avec|cuisine avec)\s+(.+)/gi
  };

  // Dictionnaire d'unités françaises
  private unitMapping: Record<string, string> = {
    'kg': 'kg', 'kilo': 'kg', 'kilos': 'kg',
    'g': 'g', 'gramme': 'g', 'grammes': 'g',
    'l': 'L', 'litre': 'L', 'litres': 'L',
    'ml': 'mL', 'millilitre': 'mL', 'millilitres': 'mL',
    'unité': 'unité(s)', 'unités': 'unité(s)', 'pièce': 'unité(s)', 'pièces': 'unité(s)',
    'boîte': 'boîte(s)', 'boîtes': 'boîte(s)',
    'paquet': 'paquet(s)', 'paquets': 'paquet(s)',
    'sachet': 'sachet(s)', 'sachets': 'sachet(s)'
  };

  constructor() {
    this.initializeRecognition();
  }

  private initializeRecognition() {
    // Support navigateur étendu
    const SpeechRecognition = 
      window.SpeechRecognition || 
      window.webkitSpeechRecognition || 
      window.mozSpeechRecognition || 
      window.msSpeechRecognition;

    if (!SpeechRecognition) {
      console.warn('🎤 Speech Recognition not supported');
      this.isSupported = false;
      return;
    }

    this.recognition = new SpeechRecognition();
    this.setupRecognitionConfig();
    this.isSupported = true;
  }

  private setupRecognitionConfig() {
    if (!this.recognition) return;

    // Configuration optimisée
    this.recognition.lang = this.config.language;
    this.recognition.continuous = this.config.continuous;
    this.recognition.interimResults = this.config.interimResults;
    this.recognition.maxAlternatives = this.config.maxAlternatives;

    // Gestion des événements avec retry logic
    this.recognition.onstart = () => {
      console.log('🎤 Voice recognition started');
      this.isListening = true;
      this.retryCount = 0;
    };

    this.recognition.onerror = (event) => {
      console.error('🎤 Voice recognition error:', event.error);
      this.isListening = false;
      
      // Retry logic intelligent
      if (this.shouldRetry(event.error)) {
        this.handleRetry();
      } else {
        this.handleError(event.error);
      }
    };

    this.recognition.onend = () => {
      console.log('🎤 Voice recognition ended');
      this.isListening = false;
    };
  }

  private shouldRetry(error: string): boolean {
    const retryableErrors = ['network', 'audio-capture', 'aborted'];
    return retryableErrors.includes(error) && this.retryCount < this.maxRetries;
  }

  private async handleRetry() {
    this.retryCount++;
    console.log(`🔄 Retrying voice recognition (${this.retryCount}/${this.maxRetries})`);
    
    // Délai progressif
    const delay = this.retryCount * 1000;
    await new Promise(resolve => setTimeout(resolve, delay));
    
    try {
      this.startListening();
    } catch (error) {
      console.error('Retry failed:', error);
    }
  }

  private handleError(error: string) {
    const errorMessages: Record<string, string> = {
      'no-speech': 'Aucun son détecté. Parlez plus fort.',
      'audio-capture': 'Erreur microphone. Vérifiez les permissions.',
      'not-allowed': 'Permission microphone refusée.',
      'network': 'Erreur réseau. Vérifiez votre connexion.',
      'service-not-allowed': 'Service de reconnaissance vocale non disponible.'
    };

    const message = errorMessages[error] || `Erreur reconnaissance vocale: ${error}`;
    toast.error(message);
  }

  async startListening(): Promise<string> {
    if (!this.isSupported) {
      throw new Error('Reconnaissance vocale non supportée');
    }

    if (this.isListening) {
      console.warn('🎤 Already listening');
      return '';
    }

    // Vérification permissions
    try {
      await navigator.mediaDevices.getUserMedia({ audio: true });
    } catch (error) {
      throw new Error('Permission microphone requise');
    }

    return new Promise((resolve, reject) => {
      if (!this.recognition) {
        reject(new Error('Recognition not initialized'));
        return;
      }

      // Timeout sécurité
      const timeout = setTimeout(() => {
        this.stopListening();
        reject(new Error('Timeout reconnaissance vocale'));
      }, 10000);

      this.recognition.onresult = (event) => {
        clearTimeout(timeout);
        
        // Traitement des résultats avec alternatives
        let bestResult = '';
        let bestConfidence = 0;

        for (let i = 0; i < event.results.length; i++) {
          const result = event.results[i];
          if (result.isFinal) {
            for (let j = 0; j < result.length; j++) {
              const alternative = result[j];
              if (alternative.confidence > bestConfidence) {
                bestConfidence = alternative.confidence;
                bestResult = alternative.transcript;
              }
            }
          }
        }

        console.log(`🎤 Best result: "${bestResult}" (confidence: ${bestConfidence})`);
        resolve(bestResult.toLowerCase().trim());
      };

      try {
        this.recognition.start();
      } catch (error) {
        clearTimeout(timeout);
        reject(error);
      }
    });
  }

  stopListening() {
    if (this.recognition && this.isListening) {
      this.recognition.stop();
      this.isListening = false;
    }
  }

  // Parser intelligent amélioré
  parseVoiceInput(transcript: string): ParsedVoiceInput {
    console.log('🧠 Parsing transcript:', transcript);
    
    const cleaned = transcript.toLowerCase().trim();
    
    // Test patterns dans l'ordre de priorité
    const patterns = [
      { regex: this.patterns.quantityWithUnit, action: 'add' as const },
      { regex: this.patterns.simpleQuantity, action: 'add' as const },
      { regex: this.patterns.remove, action: 'remove' as const },
      { regex: this.patterns.search, action: 'search' as const },
      { regex: this.patterns.productOnly, action: 'add' as const }
    ];

    for (const pattern of patterns) {
      pattern.regex.lastIndex = 0; // Reset regex
      const match = pattern.regex.exec(cleaned);
      
      if (match) {
        const result = this.extractFromMatch(match, pattern.action);
        if (result.product) {
          console.log('✅ Parsed successfully:', result);
          return result;
        }
      }
    }

    // Fallback - produit simple
    console.log('⚠️ Using fallback parsing');
    return {
      product: cleaned,
      action: 'add',
      confidence: 0.3
    };
  }

  private extractFromMatch(match: RegExpExecArray, action: 'add' | 'remove' | 'search'): ParsedVoiceInput {
    const groups = match.slice(1);
    
    if (action === 'remove' || action === 'search') {
      return {
        product: groups[0]?.trim() || '',
        action,
        confidence: 0.8
      };
    }

    // Pour les ajouts
    if (groups.length >= 3) {
      // Quantité + unité + produit
      const quantity = parseFloat(groups[0]?.replace(',', '.') || '1');
      const rawUnit = groups[1]?.toLowerCase() || '';
      const unit = this.unitMapping[rawUnit] || rawUnit;
      const product = groups[2]?.trim() || '';

      return {
        quantity: isNaN(quantity) ? 1 : quantity,
        unit,
        product,
        action,
        confidence: 0.9
      };
    } else if (groups.length >= 2) {
      // Quantité + produit
      const quantity = parseFloat(groups[0]?.replace(',', '.') || '1');
      const product = groups[1]?.trim() || '';

      return {
        quantity: isNaN(quantity) ? 1 : quantity,
        unit: 'unité(s)',
        product,
        action,
        confidence: 0.8
      };
    }

    // Produit seulement
    return {
      quantity: 1,
      unit: 'unité(s)',
      product: groups[0]?.trim() || '',
      action,
      confidence: 0.7
    };
  }

  // Test des capacités du navigateur
  getCapabilities() {
    return {
      isSupported: this.isSupported,
      hasPermission: navigator.permissions ? 
        navigator.permissions.query({ name: 'microphone' as any }) : 
        Promise.resolve({ state: 'unknown' }),
      browser: this.getBrowserInfo(),
      language: this.config.language
    };
  }

  private getBrowserInfo() {
    const ua = navigator.userAgent;
    if (ua.includes('Chrome')) return 'chrome';
    if (ua.includes('Firefox')) return 'firefox';
    if (ua.includes('Safari')) return 'safari';
    if (ua.includes('Edge')) return 'edge';
    return 'unknown';
  }
}

// Export singleton
export const voiceService = new EnhancedVoiceService();
