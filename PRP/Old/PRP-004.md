const InStoreItem = ({ item, isChecked, onCheck }) => {
  return (
    <motion.div
      initial={{ opacity: 0, x: -20 }}
      animate={{ opacity: 1, x: 0 }}
      exit={{ opacity: 0, x: 20 }}
      className={cn(
        "flex items-center gap-4 p-4 bg-white rounded-xl",
        isChecked && "opacity-50"
      )}
    >
      {/* Large Checkbox */}
      <button
        onClick={() => onCheck(item)}
        className={cn(
          "w-8 h-8 rounded-full border-2 flex items-center justify-center",
          "transition-all duration-200",
          isChecked 
            ? "bg-green-500 border-green-500" 
            : "border-gray-300"
        )}
      >
        {isChecked && <Check className="w-5 h-5 text-white" />}
      </button>
      
      {/* Item Details */}
      <div className="flex-1">
        <p className={cn(
          "font-medium text-lg",
          isChecked && "line-through text-gray-400"
        )}>
          {item.name}
        </p>
        <p className="text-sm text-gray-500">
          {item.quantity} {item.unit}
          {item.note && ` • ${item.note}`}
        </p>
      </div>
      
      {/* Price */}
      {item.estimatedPrice && (
        <div className="text-right">
          <p className="font-semibold">{item.estimatedPrice.toFixed(2)} €</p>
          {item.discount && (
            <p className="text-xs text-green-600">-{item.discount}%</p>
          )}
        </div>
      )}
    </motion.div>
  );
};
