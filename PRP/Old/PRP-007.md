tsxconst AnimatedMetricCard = ({ metric }) => {
  const [isFlipped, setIsFlipped] = useState(false);
  const animatedValue = useSpring({
    value: metric.value,
    from: { value: 0 },
    config: { tension: 280, friction: 60 }
  });
  
  return (
    <motion.div
      className="bg-white rounded-2xl p-6 shadow-sm cursor-pointer"
      whileHover={{ scale: 1.02 }}
      whileTap={{ scale: 0.98 }}
      onClick={() => setIsFlipped(!isFlipped)}
    >
      <div className="flex items-start justify-between mb-4">
        <div className="text-3xl">{metric.icon}</div>
        {metric.trend && (
          <TrendBadge value={metric.trend} positive={metric.trend > 0} />
        )}
      </div>
      
      <AnimatePresence mode="wait">
        {!isFlipped ? (
          <motion.div
            key="front"
            initial={{ rotateY: 0 }}
            exit={{ rotateY: 90 }}
          >
            <p className="text-sm text-gray-600 mb-1">{metric.title}</p>
            <animated.p className="text-3xl font-bold text-gray-900">
              {animatedValue.value.to(v => 
                typeof metric.value === 'number' 
                  ? v.toFixed(0) 
                  : metric.value
              )}
            </animated.p>
            {metric.detail && (
              <p className="text-xs text-gray-500 mt-1">{metric.detail}</p>
            )}
          </motion.div>
        ) : (
          <motion.div
            key="back"
            initial={{ rotateY: -90 }}
            animate={{ rotateY: 0 }}
          >
            <MetricDetails metric={metric} />
          </motion.div>
        )}
      </AnimatePresence>
      
      {metric.progress && (
        <div className="mt-4">
          <ProgressBar value={metric.progress} className="h-2" />
        </div>
      )}
      
      {metric.streak && (
        <StreakIndicator days={metric.value} />
      )}
    </motion.div>
  );
};
