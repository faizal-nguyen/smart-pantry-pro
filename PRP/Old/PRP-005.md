Notifications contextuelles et non-intrusives qui aident vraiment sans spammer.
🔔 DESIGN NOTIFICATIONS
typescriptinterface SmartNotificationSystem {
  // 1. TYPES DE NOTIFICATIONS
  categories: {
    'expiry-alerts': {
      priority: 'high';
      timing: 'morning'; // 8h-9h
      frequency: 'daily';
      examples: [
        "🥛 Votre lait expire demain",
        "🍓 3 produits expirent cette semaine"
      ];
    };
    
    'meal-suggestions': {
      priority: 'medium';
      timing: 'before-meals'; // 11h30, 18h30
      frequency: 'contextual';
      examples: [
        "🍝 Avec vos ingrédients, essayez ces pâtes carbonara",
        "🥗 Il fait chaud, que diriez-vous d'une salade ?"
      ];
    };
    
    'shopping-reminders': {
      priority: 'low';
      timing: 'location-based'; // Près d'un supermarché
      frequency: 'weekly';
      examples: [
        "🛒 Vous passez près de Carrefour, 5 articles sur votre liste",
        "📍 C'est samedi ! Jour de courses habituel"
      ];
    };
    
    'achievement-celebration': {
      priority: 'low';
      timing: 'immediate';
      frequency: 'on-trigger';
      examples: [
        "🎉 Bravo ! 0 gaspillage cette semaine",
        "💰 Vous avez économisé 23€ ce mois"
      ];
    };
  };

  // 2. SMART SCHEDULING
  scheduling: {
    doNotDisturb: {
      enabled: true;
      hours: { start: '22:00', end: '08:00' };
      exceptions: ['critical-expiry'];
    };
    
    batching: {
      enabled: true;
      strategy: 'group-similar';
      maxBatch: 3;
    };
    
    contextual: {
      mealTimes: true; // Adapte aux heures de repas
      location: true; // Utilise la géolocalisation
      habits: true; // Apprend de vos habitudes
    };
  };
}

tsxconst SmartNotification = ({ notification }) => {
  const [isExpanded, setIsExpanded] = useState(false);
  
  return (
    <motion.div
      initial={{ opacity: 0, y: -20, scale: 0.9 }}
      animate={{ opacity: 1, y: 0, scale: 1 }}
      exit={{ opacity: 0, x: 100 }}
      className="bg-white rounded-2xl shadow-lg p-4 mb-3"
    >
      <div className="flex items-start gap-3">
        {/* Icon avec animation */}
        <motion.div
          animate={{ rotate: [0, 10, -10, 0] }}
          transition={{ duration: 0.5 }}
          className="text-2xl"
        >
          {notification.icon}
        </motion.div>
        
        {/* Content */}
        <div className="flex-1">
          <h4 className="font-semibold text-gray-900">
            {notification.title}
          </h4>
          <p className="text-sm text-gray-600 mt-1">
            {notification.message}
          </p>
          
          {/* Quick Actions */}
          {notification.actions && (
            <div className="flex gap-2 mt-3">
              {notification.actions.map(action => (
                <button
                  key={action.id}
                  onClick={action.handler}
                  className={cn(
                    "px-3 py-1.5 rounded-lg text-sm font-medium",
                    action.primary 
                      ? "bg-blue-500 text-white" 
                      : "bg-gray-100 text-gray-700"
                  )}
                >
                  {action.label}
                </button>
              ))}
            </div>
          )}
        </div>
        
        {/* Dismiss */}
        <button
          onClick={() => dismissNotification(notification.id)}
          className="text-gray-400 hover:text-gray-600"
        >
          <X className="w-4 h-4" />
        </button>
      </div>
    </motion.div>
  );
};
